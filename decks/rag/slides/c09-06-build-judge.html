from typing import Literal
from google import genai
from google.genai import types
from pydantic import BaseModel
from sklearn.metrics import cohen_kappa_score

class ClaimVerdict(BaseModel):
    claim: str
    verdict: Literal["supported", "unsupported", "contradicted"]
    evidence_quote: str          # must be copied verbatim from context, or "" if none
class JudgeOutput(BaseModel):
    claims: list[ClaimVerdict]

JUDGE_PROMPT = ("You are a refund-policy QA auditor. Split ANSWER into atomic factual claims. "
                "For each claim decide: supported (CONTEXT states it), unsupported (CONTEXT is silent), "
                "contradicted (CONTEXT says otherwise). Quote the exact CONTEXT sentence as evidence_quote. "
                "Judge only against CONTEXT.\nCONTEXT:\n{context}\n\nANSWER:\n{answer}")
client = genai.Client()                        # reads GEMINI_API_KEY

def judge(context: str, answer: str, model: str) -> list[ClaimVerdict]:
    resp = client.models.generate_content(model=model, contents=JUDGE_PROMPT.format(context=context, answer=answer),
        config=types.GenerateContentConfig(temperature=0.0, response_mime_type="application/json",
                                           response_schema=JudgeOutput))
    out: JudgeOutput = resp.parsed
    for c in out.claims:                       # anti-hallucination check on the judge itself
        if c.verdict == "supported" and c.evidence_quote not in context:
            c.verdict = "unsupported"
    return out.claims

# part 1: imports, ClaimVerdict, JudgeOutput, JUDGE_PROMPT, client, judge()
POLICY = ["Physical goods may be returned within 14 days of delivery for a full refund.",
          "Digital gift cards are non-refundable once the code has been revealed.",
          "Refunds are issued to the original payment method within 5 business days."]
CTX = " ".join(POLICY)                         # the three ShopCo refund-policy chunks
GOLD = [("Physical goods can be returned within 14 days of delivery.", "supported"),
        ("Refunds go to the original payment method within 5 business days.", "supported"),
        ("Revealed gift cards can be refunded.", "contradicted"),
        ("Return shipping is free for all orders.", "unsupported"),
        ("Gift cards are non-refundable once the code is revealed.", "supported"),
        ("Refund requests must be filed within 30 days.", "unsupported")]   # use >= 100 per domain
for model in ["gemini-3.5-flash-lite", "gemini-3.8-flash"]:
    pred = [judge(CTX, a, model)[0].verdict for a, _ in GOLD]
    human = [h for _, h in GOLD]
    bad = [p for p, h in zip(pred, human) if h != "supported"]
    admit = sum(p == "supported" for p in bad) / len(bad)
    print(f"{model:24} kappa={cohen_kappa_score(human, pred):5.2f}  unsupported-admit-rate={admit:4.0%}")

This build-up adds one feature to the ShopCo gate: proof that the faithfulness judge agrees with humans. The judge returns one of three verdicts per claim with a verbatim evidence quote, and a deterministic check downgrades any "supported" verdict whose quote is not in the context. We then run two Gemini models on six human-labelled refund claims and report Cohen's kappa and the unsupported-admit rate; in this illustrative run flash-lite waves one unsupported claim through, which is exactly the error that ships a "gift cards are refundable" answer. Analogy: you calibrate a scale with known weights before you use it to price goods. The papers back this up: AGO measured a cheap judge at AUROC 0.603 against 0.783, and Jev-as-a-Judge found two verifiers with equal accuracy but different unsupported-admit rates. Likely question: "Is six claims enough?" No, it only shows the method; use at least 100 human-labelled claims per domain and re-run calibration whenever the model id, the prompt or the temperature changes.
