from langchain_classic.retrievers import ParentDocumentRetriever
from langchain_core.documents import Document
from langchain_core.embeddings import Embeddings
from langchain_core.stores import InMemoryStore
from langchain_core.vectorstores import InMemoryVectorStore
from langchain_text_splitters import RecursiveCharacterTextSplitter
from sentence_transformers import SentenceTransformer
from kb import DOCS, QUERIES, DOC_OF, doc_text

class STEmbeddings(Embeddings):                            # 6-line adapter, avoids langchain-huggingface pin
    def __init__(self, name: str):
        self.m = SentenceTransformer(name)
    def embed_documents(self, texts: list[str]) -> list[list[float]]:
        return self.m.encode_document(texts, normalize_embeddings=True).tolist()
    def embed_query(self, text: str) -> list[float]:
        return self.m.encode_query(text, prompt="Represent this sentence for searching relevant passages: ",
                                   normalize_embeddings=True).tolist()

retriever = ParentDocumentRetriever(
    vectorstore=InMemoryVectorStore(STEmbeddings("BAAI/bge-small-en-v1.5")),
    docstore=InMemoryStore(),
    child_splitter=RecursiveCharacterTextSplitter(chunk_size=160, chunk_overlap=0, separators=["\n", ". "]),
    search_kwargs={"k": 12},                                # BEFORE: missing -> similarity_search k=4
)
parents = [Document(page_content=doc_text(d), metadata={"doc_id_biz": d}) for d in DOCS]
retriever.add_documents(parents, ids=list(DOCS))            # parent ids = D1..D4
hits = mrr_sum = 0.0
for qid, (text, gold_chunks) in QUERIES.items():
    gold_docs = {DOC_OF[c] for c in gold_chunks}
    got = [d.metadata["doc_id_biz"] for d in retriever.invoke(text)]
    hits += bool(set(got[:1]) & gold_docs)
    mrr_sum += next((1 / r for r, d in enumerate(got, 1) if d in gold_docs), 0.0)
    print(f"{qid} gold={sorted(gold_docs)} got={got}")
print(f"parent-level Hit@1={hits / len(QUERIES):.3f}  MRR={mrr_sum / len(QUERIES):.3f}")

Parent-child (small-to-big) retrieval embeds small children for a precise match and hands the model the whole parent, here the full Northwind policy document, so q11 gets the 60-day loyalty window and the 5-7 day refund rule together and c07 arrives under its "Warranty Policy: Tents" title. The trap is the order of operations: ParentDocumentRetriever runs similarity_search for k children first and only then de-duplicates to parents, and InMemoryVectorStore defaults to k=4, so four children from one policy collapse into one parent; ask for about three children per parent you want. Install langchain-classic, not langchain-huggingface[full], because that extra pins sentence-transformers below 6 and pip downgrades or fails. The price is context: each parent is 5-10 times the tokens of a child, and the October 2026 ablation 2610.10170 found that naive two-stage hierarchical retrieval hurts, while structure-aligned chunks help, so make the parent a real policy section and cap it so 3-5 parents fit the budget. Analogy: you search the index cards, but you hand the reader the whole chapter. Likely question: "Why not just use bigger chunks?" Big chunks blur the embedding; small children keep the match sharp, and the parent keeps the answer complete. Next: three papers that move this kind of work to index time.
