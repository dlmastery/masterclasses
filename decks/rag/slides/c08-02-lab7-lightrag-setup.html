import asyncio, os
import numpy as np
from lightrag import LightRAG, QueryParam
from lightrag.llm.gemini import gemini_model_complete, gemini_embed
from lightrag.utils import wrap_embedding_func_with_attrs

MODEL = os.getenv("MODEL", "gemini-3.8-flash")
WORKING_DIR = "./claims_kg"
DOCS = {
    "claims/storm-2026-041.txt": "Claim 2026-041: hail damaged the roof of 12 Elm St "
                                 "on 2026-05-03. Adjuster Maria Lopez approved USD 18,400.",
    "claims/storm-2026-057.txt": "Claim 2026-057: hail damaged a car at 14 Elm St "
                                 "on 2026-05-03. Adjuster Maria Lopez flagged possible fraud.",
    "policy/hail-guidelines.txt": "Hail claims from the same street and date must be reviewed "
                                  "together by the Special Investigations Unit (SIU).",
}

async def llm_model_func(prompt, system_prompt=None, history_messages=None, **kwargs):
    return await gemini_model_complete(
        prompt, system_prompt=system_prompt, history_messages=history_messages or [],
        model_name=MODEL, **kwargs)

@wrap_embedding_func_with_attrs(embedding_dim=1536, send_dimensions=True, max_token_size=2048,
                                model_name="gemini-embedding-001")
async def embedding_func(texts: list[str], embedding_dim: int | None = None,
                         context: str = "document") -> np.ndarray:
    return await gemini_embed.func(texts, model="gemini-embedding-001",
                                   embedding_dim=embedding_dim, context=context)
# part 2, main(): LightRAG(..., llm_model_name=MODEL, embedding_func=embedding_func)

This is the run-it-Monday slide of Lab 7: pin lightrag-hku 1.5.7 and google-genai 2.33.0, and export GEMINI_API_KEY, because the LightRAG Gemini binding does not read GOOGLE_API_KEY. The code is adapted from the official lightrag_gemini_demo.py at v1.5.7, with two fixes: a current model instead of the shut-down gemini-2.0-flash, and an embedding function that accepts the embedding_dim and context arguments that LightRAG injects. Walk the three bands: the three Harbor Mutual files are keyed by path, so every answer can cite them; the LLM wrapper passes model_name, and part 2 also sets llm_model_name, because the binding raises when it finds neither; the decorator declares 1536 dimensions and sends them as output_dimensionality, the binding L2-normalises below 3072, and the context parameter lets queries and documents embed differently. Analogy: the decorator is a shipping label, and LightRAG reads the label to size the vector store before any vector arrives. Likely question: "Can I switch to a 3072-dim embedder later?" Not in the same working_dir: the stored vectors keep the old dimension, so use a new directory or re-index, which is the gotcha later in this chapter.
