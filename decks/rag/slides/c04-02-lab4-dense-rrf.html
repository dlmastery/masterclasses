# lab3_hybrid.py -- dense (BGE) + BM25, fused with Reciprocal Rank Fusion
from sentence_transformers import SentenceTransformer
from kb import IDS, TEXT
from evalkit import evaluate
from lab2_bm25 import bm25_search

model = SentenceTransformer("BAAI/bge-small-en-v1.5")
BGE_QUERY = "Represent this sentence for searching relevant passages: "
doc_emb = model.encode_document([TEXT[i] for i in IDS], normalize_embeddings=True)

def dense_search(query: str, k: int = 10) -> list[str]:
    q_emb = model.encode_query(query, prompt=BGE_QUERY, normalize_embeddings=True)
    scores = model.similarity(q_emb, doc_emb)[0]            # cosine, shape (n_docs,)
    top = scores.argsort(descending=True)[:k].tolist()
    return [IDS[i] for i in top]

def rrf(rankings: list[list[str]], k: int = 60, weights: list[float] | None = None) -> list[str]:
    weights = weights or [1.0] * len(rankings)
    fused: dict[str, float] = {}
    for w, ranking in zip(weights, rankings):
        for rank, doc_id in enumerate(ranking, start=1):
            fused[doc_id] = fused.get(doc_id, 0.0) + w / (k + rank)
    return sorted(fused, key=fused.get, reverse=True)

def hybrid_search(query: str, depth: int = 20, k_rrf: int = 60) -> list[str]:
    return rrf([bm25_search(query, depth), dense_search(query, depth)], k=k_rrf)

if __name__ == "__main__":
    evaluate("BM25", bm25_search)
    evaluate("dense (bge-small)", dense_search)
    evaluate("hybrid RRF k=60", hybrid_search)
    evaluate("hybrid RRF k=2", lambda q: hybrid_search(q, k_rrf=2))
    evaluate("hybrid RRF dense x2", lambda q: rrf([bm25_search(q, 20), dense_search(q, 20)], weights=[1, 2]))

Each list adds w / (k + rank). Fuse ranks, not scores.

Depth 20 per list. Fusion cannot rescue a chunk below depth.

Dense fixes paraphrases, BM25 fixes "P-220". R@5 gains more than R@1.

Walk the four bands: encode_query only adds the BGE instruction if you pass prompt=, because the model ships no prompts config; rrf() is the Cormack et al. SIGIR 2009 formula w / (k + rank) with an optional weight per list; depth 20 sets what each list can contribute; and the five evaluate() rows go through the same Chapter 1 harness. The animation is query q13 with illustrative ranks: BM25 puts the gold chunk c18 at rank 7 because it says "import duties", dense puts it first, and the sum 1/67 + 1/61 lifts it to fused rank 4, inside the top 5 but not at rank 1. Analogy: RRF is a vote where each list gives more points to its top picks, and a chunk both lists like wins. Likely question: "Why k=60?" It is the paper's value and it keeps the curve flat, so agreement between lists matters more than one list's first place; the output shows k=2 dropping q11, and the next slide shows that Qdrant defaults to k=2.
