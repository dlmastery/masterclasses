from functools import lru_cache
from google import genai
from google.genai import types
from pydantic import BaseModel, Field
from evalkit import evaluate
from lab3_hybrid import dense_search, hybrid_search, rrf
client = genai.Client()                     # reads GEMINI_API_KEY / GOOGLE_API_KEY
class Rewrites(BaseModel):
    paraphrases: list[str] = Field(description="3 rewordings using policy vocabulary")
    hypothetical_answer: str = Field(
        description="2-sentence passage that would answer the question")
    sub_questions: list[str] = Field(
        description="1-3 self-contained sub-questions; 1 if the question is simple")
PROMPT = ("You rewrite customer-support questions for search over an outdoor retailer's "
          "policy KB (returns, refunds, warranty, payment error codes, shipping). "
          "Keep error codes exactly as written.\nQuestion: {q}")

@lru_cache(maxsize=1024)                    # never pay twice for the same query
def rewrite(query: str) -> Rewrites:
    resp = client.models.generate_content(
        model="gemini-3.8-flash", contents=PROMPT.format(q=query),
        config=types.GenerateContentConfig(
            response_mime_type="application/json", response_schema=Rewrites,
            thinking_config=types.ThinkingConfig(thinking_level="LOW")))
    return resp.parsed

def fusion_search(query: str) -> list[str]:
    rw = rewrite(query)
    lists = [hybrid_search(query)]          # always keep the original query
    lists += [hybrid_search(p) for p in rw.paraphrases + rw.sub_questions]
    lists.append(dense_search(rw.hypothetical_answer, 20))  # HyDE: dense only
    return rrf(lists, k=60, weights=[2.0] + [1.0] * (len(lists) - 1))
evaluate("hybrid", hybrid_search)
evaluate("RAG-Fusion (MQ+HyDE+decomp)", fusion_search)

This build fixes q11, "loyalty member: how long is the return window and when is the refund paid?", which needs two chunks: c04 for the 60-day loyalty window and c02 for the 5-7 day refund. One Gemini 3.8 Flash call with a Pydantic response_schema and the JSON MIME type returns three paraphrases, a two-sentence hypothetical answer and one to three sub-questions; thinking is LOW and an lru_cache means a repeated question costs nothing. fusion_search keeps the original query at weight 2.0, because the September SPLADE-v3 study (2609.37911) found gains in every setting that keeps at least 30% of the weight on the original query, and it sends the HyDE passage to the dense side only, because SAFE-QE (2610.13810) counted 57.5 invented terms per query for HyDE. The illustrative result moves R@3 from 0.867 to 0.933 and R@5 from 0.933 to 0.967, at the cost of one LLM call and five to seven retrievals per query; the ranks in the animation are illustrative too. Analogy: you ask the question three ways and split it in two, but you still listen most to the customer's own words. Likely question: "Should every query go through this?" No: gate it to multi-part, vague or very short queries, skip it when the query holds an exact code like E-4012, check that every CODE.findall(query) code survives in each rewrite, and fall back to the original query when resp.parsed is None.
