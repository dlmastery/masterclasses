# parts 1-2: imports, llm, store, Plan, Hop, State, plan(), hop(), next_step()
def answer(s: State):
    open_q = [r["q"] for r in s["register"] if r["a"] is None]
    if open_q:                                                   # sufficiency gate
        return {"answer": f"Cannot answer yet; unresolved: {open_q}"}
    chain = "\n".join(f"{r['q']} -> {r['a']} [{r['src']}]" for r in s["register"])
    prompt = (f"Using only this evidence chain, answer: {s['question']}\n"
              f"{chain}\nCite every id used.")         # answers only, no passage text
    return {"answer": llm.invoke(prompt).text}
g = StateGraph(State)
g.add_node(plan); g.add_node(hop); g.add_node(answer)
g.add_edge(START, "plan"); g.add_edge("plan", "hop")
g.add_conditional_edges("hop", next_step, ["hop", "answer"])   # path list
g.add_edge("answer", END)
app = g.compile()
if __name__ == "__main__":
    q = ("Who must approve a return of the jacket made by the supplier "
         "that failed the March 2026 audit?")
    out = app.invoke({"question": q}, {"recursion_limit": 20})   # worst case: 10 steps
    for r in out["register"]:
        print(r)
    print("\nANSWER:", out["answer"])
