# part 1 (step 1 of 4): imports, sys, PROJECT, LOC, client, corpus, import_files
store = gt.VertexRagStore(
    rag_resources=[gt.VertexRagStoreRagResource(rag_corpus=corpus.name)])
cfg = gt.RagRetrievalConfig(
    top_k=5, filter=gt.RagRetrievalConfigFilter(vector_distance_threshold=0.5))
q = "How many weeks of paid parental leave do employees in Germany get?"
ctx = client.rag.retrieve_contexts(
    vertex_rag_store=store, query=types.RagQuery(text=q, rag_retrieval_config=cfg))
for c in ctx.contexts.contexts:
    print(f"dist={c.score:.3f}  {c.source_display_name}  {c.text[:70]!r}")

rerank_cfg = gt.RagRetrievalConfig(top_k=5, ranking=gt.RagRetrievalConfigRanking(
    rank_service=gt.RagRetrievalConfigRankingRankService(
        model_name="semantic-ranker-512@latest")))
tool = gt.Tool(retrieval=gt.Retrieval(vertex_rag_store=gt.VertexRagStore(
    rag_resources=store.rag_resources, rag_retrieval_config=rerank_cfg)))
gen = genai.Client(enterprise=True, project=PROJECT, location=LOC)
resp = gen.models.generate_content(model="gemini-3.8-flash", contents=q,
    config=gt.GenerateContentConfig(tools=[tool], temperature=0))
print("\nANSWER:", resp.text)
for ch in resp.candidates[0].grounding_metadata.grounding_chunks or []:
    print("  source:", ch.retrieved_context.title, ch.retrieved_context.uri)

if "--cleanup" in sys.argv:
    client.rag.delete_corpus(name=corpus.name)

This is the second half of the GlobalCorp script: we retrieve with a distance threshold, look at the raw contexts, then let Gemini answer with a reranked retrieval tool. The key trap is the threshold: RAG Engine returns a distance, so 0.212 is better than 0.471, and anything above 0.5 is dropped. Think of it as a radius around the question, not a minimum score. We print the raw chunk text because the eval harness from chapter 9 must score exactly what the model saw; for hybrid search you can also set RagQueryRanking alpha, where 0 is sparse only and 1 is dense only. A likely question is "why rerank if the filter already cut the list?": the grounded call uses top 5 plus the ranking API, which costs under 100 ms, while an LLM ranker costs 1 to 2 seconds.
