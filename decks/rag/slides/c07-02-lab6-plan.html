import os
from typing import TypedDict
from pydantic import BaseModel, Field
from langchain_core.documents import Document
from langchain_core.vectorstores import InMemoryVectorStore
from langchain_google_genai import ChatGoogleGenerativeAI, GoogleGenerativeAIEmbeddings
from langgraph.graph import StateGraph, START, END
llm = ChatGoogleGenerativeAI(model=os.getenv("MODEL", "gemini-3.8-flash"))
PASSAGES = [
    ("AUD-1", "Kestrel Textiles failed the March 2026 audit and is on audit hold."),
    ("AUD-2", "Bluefin Gear passed the March 2026 audit with no findings."),
    ("SUP-4", "Kestrel Textiles manufactures the TrailLite rain jacket."),
    ("SUP-5", "Bluefin Gear manufactures the Summit fleece."),
    ("RET-2", "Returns from a supplier on audit hold need the Quality Lead, Priya Natarajan."),
    ("RET-1", "Standard returns are approved automatically by the returns portal.")]
store = InMemoryVectorStore(GoogleGenerativeAIEmbeddings(model="gemini-embedding-001"))
store.add_documents([Document(t, metadata={"id": i}) for i, t in PASSAGES])
class Plan(BaseModel):
    subquestions: list[str] = Field(description=
        "2-4 single-fact sub-questions in order; refer to earlier answers as #1, #2")
class Hop(BaseModel):
    answer: str = Field(description="Short answer, or UNKNOWN if the passages do not state it")
    source_id: str = Field(description="Id of the passage that states the answer, or ''")
class State(TypedDict, total=False):       # total=False: a key may be missing
    question: str; register: list[dict]; hops: int; answer: str   # row: {q, a, src, tries}
def plan(s: State):
    p = llm.with_structured_output(Plan).invoke(
        f"Decompose into single-fact sub-questions: {s['question']}")
    return {"register": [{"q": q, "a": None, "src": None, "tries": 0}
                         for q in p.subquestions], "hops": 0}

Walk the three bands: six one-fact passages, where the answer needs AUD-1, SUP-4 and RET-2 together; the Plan and Hop schemas, where the Field description shows the model the #1 placeholder format, and the State TypedDict with total=False, which means a key may be missing until a node writes it; and plan(), which writes a register of open rows and sets hops to 0. Your model may word the sub-questions differently from the chips on the slide; if it drops the #1 placeholder, the substitution on the next slide silently does nothing, so print the plan first. This is my own implementation of the training-free requirement register from arXiv 2610.21184, not the paper's code. Analogy: a detective writes every open question on the case board before the first interview, and leaves a blank for each name that is not known yet. Likely question: "Why plan up front instead of letting the agent search freely?" Because free search drifts away from the open sub-question. Next: hop() resolves the rows one at a time.
