# lab7_photos.py -- claim photos and text questions in one embedding space
import numpy as np
from google import genai
from google.genai import types

client = genai.Client()                                    # reads GEMINI_API_KEY
PHOTOS = ["photos/2026-041-roof.png", "photos/2026-057-car.png"]

def embed_image(path: str) -> np.ndarray:
    with open(path, "rb") as f:
        img = f.read()
    r = client.models.embed_content(
        model="gemini-embedding-2",
        contents=[types.Part.from_bytes(data=img, mime_type="image/png")],
        config=types.EmbedContentConfig(output_dimensionality=768))
    return np.asarray(r.embeddings[0].values, dtype=np.float32)

def embed_query(q: str) -> np.ndarray:                    # no task_type on gemini-embedding-2
    r = client.models.embed_content(
        model="gemini-embedding-2",
        contents=[types.Content(parts=[types.Part.from_text(text=f"task: search result | query: {q}")])],
        config=types.EmbedContentConfig(output_dimensionality=768))   # auto-normalised
    return np.asarray(r.embeddings[0].values, dtype=np.float32)

P = np.stack([embed_image(p) for p in PHOTOS])                # (2, 768), unit rows
scores = P @ embed_query("hail dents on a car bonnet")        # dot product == cosine
for i in np.argsort(-scores):
    print(f"{scores[i]:.2f}  {PHOTOS[i]}")

One image Part per call. 768 dims.

No task_type: put the task in the text. One Content per input.

Truncated dims are auto-normalised, so dot product = cosine.

Rank the photos by score for one text question.

This is the last step of Lab 7: the Harbor Mutual investigator types a text question and gets back the claim photo that matches it. Walk the four bands: one image Part per call at 768 dimensions; the query carries its task as a text prefix because gemini-embedding-2 has no task_type, and each input gets its own Content because several inputs in one contents list give one aggregated vector; truncated dimensions are auto-normalised, so the dot product is the cosine; then the ranking. Analogy: the photo and the question are two compass needles of the same length, and the score is only the angle between them. Likely question: "Can I add these photo vectors to the Lab 7 LightRAG index?" No. That index uses gemini-embedding-001, and its vectors are not compatible with gemini-embedding-2, so keep a separate index or re-embed everything.
