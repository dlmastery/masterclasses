# lab3_splitters_refund.py
from langchain_text_splitters import (
    MarkdownHeaderTextSplitter, RecursiveCharacterTextSplitter)
from transformers import AutoTokenizer

REFUND_MD = open("data/refund_policy.md", encoding="utf-8").read()  # 3 sections, 5 rules
tok = AutoTokenizer.from_pretrained("Qwen/Qwen3-Embedding-0.6B")
ntok = lambda s: len(tok.tokenize(s))

# A) naive: fixed character windows over the whole file
naive = RecursiveCharacterTextSplitter(
    chunk_size=200, chunk_overlap=0).split_text(REFUND_MD)

# B) structure first: split on headers, keep header text, then size by TOKENS
headers = [("#", "h1"), ("##", "h2"), ("###", "h3")]
sections = MarkdownHeaderTextSplitter(
    headers_to_split_on=headers, strip_headers=False).split_text(REFUND_MD)
sizer = RecursiveCharacterTextSplitter.from_huggingface_tokenizer(
    tok, chunk_size=64, chunk_overlap=8, add_start_index=True)
structured = sizer.split_documents(sections)

orphans = sum(1 for c in naive if not c.lstrip().startswith("#"))
print(f"naive: {len(naive)} chunks, {orphans} start without a heading")
print(f"structured: {len(structured)} chunks")
for d in structured:
    path = " > ".join(d.metadata[k] for k in ("h1", "h2", "h3") if k in d.metadata)
    print(f"  S | {ntok(d.page_content):>3} tok | "
          f"start={d.metadata['start_index']:>3} | {path}")
chunk_size counts characters (len). Base defaults: 4000 / 200.
Header split adds h1/h2/h3 metadata. strip_headers=False keeps headings in the embedded text.
chunk_size=64 now counts embedder tokens. Gotcha: start_index is per section, not per file.
Start at 256–512 embedder tokens. Then tune on your eval set.

This is the last step of Lab 2: the refund and cancellation policy is Markdown, so we compare a naive character splitter with a structure-first splitter. The naive splitter cuts every 200 characters, and three of its four chunks start in the middle of a section, so the free-look rule "No claim must have been filed" lands in a chunk about annual policies. Think of a bread slicer that cuts at fixed widths and ignores where one sandwich ends and the next begins. The structured version first splits at the headings, keeps the heading text, and then sizes each section in Qwen3 tokens, so every chunk carries its own heading path as metadata for filters and citations. A likely question is why start_index is 0 everywhere: after split_documents the offset is relative to the section, so store the section anchor too if the UI must highlight characters in the original file.
