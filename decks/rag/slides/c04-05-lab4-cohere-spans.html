# lab6_api_rerank.py -- Cohere Rerank v5 (+ relevant spans) and Voyage rerank-3
import cohere
import voyageai
from cohere import RerankRelevantSpans
from kb import TEXT
from evalkit import evaluate
from lab3_hybrid import hybrid_search
co = cohere.ClientV2()                    # reads CO_API_KEY
vo = voyageai.Client()                    # reads VOYAGE_API_KEY
def cohere_search(query: str, candidates: int = 20) -> list[str]:
    cand_ids = hybrid_search(query)[:candidates]
    resp = co.rerank(model="rerank-v5.0-pro", query=query, documents=[TEXT[i] for i in cand_ids],
                     top_n=10, max_tokens_per_doc=1024)
    return [cand_ids[r.index] for r in resp.results]
def voyage_search(query: str, candidates: int = 20) -> list[str]:
    cand_ids = hybrid_search(query)[:candidates]
    out = vo.rerank(query, [TEXT[i] for i in cand_ids], model="rerank-3", top_k=10)
    return [cand_ids[r.index] for r in out.results]
def compressed_context(query: str, keep: int = 3) -> list[str]:
    cand_ids = hybrid_search(query)[:20]
    docs = [TEXT[i] for i in cand_ids]
    resp = co.rerank(model="rerank-v5.0-pro", query=query, documents=docs, top_n=keep,
                     relevant_spans=RerankRelevantSpans(mode="enabled", threshold=0.3, max_spans=2))
    out = []
    for r in resp.results:
        spans = r.relevant_spans or []
        text = " ... ".join(docs[r.index][s.start:s.end] for s in spans) or docs[r.index]
        out.append(f"[{cand_ids[r.index]} {r.relevance_score:.2f}] {text}")
    return out
if __name__ == "__main__":
    q11 = "loyalty member: how long is the return window and when is the refund paid?"
    print("\n".join(compressed_context(q11)))
    evaluate("hybrid + Cohere v5 pro", cohere_search)
    evaluate("hybrid + Voyage rr-3", voyage_search)

This is the last step of Lab 4: the same 20 hybrid candidates go to two hosted rerankers, Cohere Rerank v5 and Voyage rerank-3. First point: both APIs return an index into the list you sent, so map it back with cand_ids[r.index], as in step 3, or every metric falls to the floor. The new part is relevant_spans: Cohere v5 marks the character ranges that answer the query, and slicing them drops the sentences the LLM does not need, with no second LLM call. Think of it as a highlighter pen: the reranker reads the page once and gives you only the marked lines. The numbers are illustrative and come from 15 queries, so do not call one vendor better from this run. A likely question is what happens when no span passes the threshold: relevant_spans is empty, and the code falls back to the full chunk.
