# steps 1-3: imports, MODEL, HR_DOCS, router, route_on_category, refuse, answerer, verifier, gate
from lab5_it_triage import it_triage              # step 4: the IT branch
MAX_ATTEMPTS = 3                                  # the budget lives in code, not in the prompt
class Grade(BaseModel):                           # 3-way grade, as in arXiv 2610.22652
    label: Literal["correct", "ambiguous", "incorrect"] = Field(
        description="correct: evidence fully answers; ambiguous: partly; incorrect: irrelevant")
    missing: str = Field(description="What the evidence is missing, or ''")
def take_question(node_input: str):               # now also seeds query and attempts
    return Event(state={"question": node_input, "query": node_input, "attempts": 0})
def retrieve_hr(query: str, attempts: int):       # replaces retrieve_hr(question, k=2)
    q = set(re.findall(r"[a-z]+", query.lower()))
    scored = sorted(((len(q & set(re.findall(r"[a-z]+", t.lower()))), d, t)
                     for d, t in HR_DOCS.items()), reverse=True)
    hits = [f"[{d}] {t}" for s, d, t in scored[:2] if s >= 2]   # can return nothing
    return Event(state={"evidence": "\n".join(hits) or "(no results)", "attempts": attempts + 1})
grader = Agent(name="grader", model=MODEL, output_schema=Grade, output_key="grade",
               instruction="Question: {question}\nEvidence:\n{evidence}\nGrade the evidence.")
def decide(node_input: Grade, attempts: int):
    if node_input.label == "correct":
        yield Event(route="answer")
    elif attempts >= MAX_ATTEMPTS:
        yield Event(route="abstain")
    else:
        yield Event(route="rewrite", state={"missing": node_input.missing})
rewriter = Agent(name="rewriter", model=MODEL, output_key="query",
    instruction="Query '{query}' failed; missing: {missing}. Write ONE better keyword "
                "query for an HR policy manual for: {question}. Output only the query.")
def abstain(question: str, attempts: int):
    yield Event(message=f"Opened an HR ticket after {attempts} searches: '{question}'.")
root_agent = Workflow(name="hr_helpdesk", edges=[
    ("START", take_question, router, route_on_category),
    (route_on_category, {"hr_policy": retrieve_hr, "it_support": it_triage, "out_of_scope": refuse}),
    (retrieve_hr, grader, decide),
    (decide, {"answer": answerer, "rewrite": rewriter, "abstain": abstain}),
    (answerer, verifier, gate),                   # the verifier gate from step 3 stays
    (rewriter, retrieve_hr),                      # the corrective back-edge
])

Predicting Partial Answer Quality and Utility in Agentic RAG

Answer quality plateaus early. Early stopping cut about 11% of the iterations at about 98% of the quality.
Keep a small, hard MAX_ATTEMPTS and an abstain branch.
This build adds Corrective RAG style self-correction to the HR branch of the Northwind helpdesk, in the same lab5_helpdesk.py: take_question now also seeds query and attempts, and retrieve_hr now reads query instead of question. Step 1: the first try loops decide straight back to retrieve_hr, and ADK rejects it when it builds the graph, because a cycle needs at least one routed edge. Steps 2 and 3: retrieve_hr can return "(no results)" and counts the attempt; the grader gives a 3-way grade; decide mixes that grade with a hard budget of 3 kept in code; the rewriter's output_key overwrites state["query"], so "Lisbon for six weeks" becomes "working abroad over 30 days HR approval" and attempt 2 finds REMOTE-2026, which still goes through the verifier gate. Analogy: a librarian who rephrases your request at most three times and then opens a ticket for a human. The gotcha is real: pass the whole grade dict into the rewriter prompt and the loop never recovers, so pass only the missing part. Likely question: "Why 3 and not 10?" The September paper shows quality plateaus early: early stopping cut about 11% of iterations and kept about 98% of quality. Next: the three ADK errors you hit while you wire this graph.
