import json, os, anthropic
from kb import CHUNKS, doc_text
from evalkit import evaluate
client = anthropic.Anthropic()
DOCUMENT_CONTEXT_PROMPT = "\n<document>\n{doc_content}\n</document>\n"
CHUNK_CONTEXT_PROMPT = """
Here is the chunk we want to situate within the whole document
<chunk>
{chunk_content}
</chunk>

Please give a short succinct context to situate this chunk within the overall document \
for the purposes of improving search retrieval of the chunk.
Answer only with the succinct context and nothing else.
"""
def situate(doc: str, chunk: str) -> str:
    resp = client.messages.create(
        model="claude-haiku-5-5", max_tokens=300,
        output_config={"effort": "low"},                  # no temperature: 400 on API 2026-10-21
        messages=[{"role": "user", "content": [
            {"type": "text", "text": DOCUMENT_CONTEXT_PROMPT.format(doc_content=doc),
             "cache_control": {"type": "ephemeral"}},     # same doc for every chunk -> cache it
            {"type": "text", "text": CHUNK_CONTEXT_PROMPT.format(chunk_content=chunk)}]}])
    u = resp.usage
    print(f"  cache_write={u.cache_creation_input_tokens}"
          f" cache_read={u.cache_read_input_tokens} in={u.input_tokens}")
    return next(b.text for b in resp.content if b.type == "text").strip()
CACHE = "contexts.json"
ctx = json.load(open(CACHE)) if os.path.exists(CACHE) else {}
for cid, did, text in CHUNKS:                              # document order keeps the cache warm
    if cid not in ctx:
        ctx[cid] = situate(doc_text(did), text)
json.dump(ctx, open(CACHE, "w"), indent=1)
ctext = [f"{ctx[cid]}\n{text}" for cid, _, text in CHUNKS]  # index this for BM25 AND dense

Query q06 asks when the tent warranty starts, but chunk c07 only says "the window starts on the purchase date", so BM25 and the embedder both rank it low. Contextual Retrieval fixes the chunk, not the query: Claude Haiku 5.5 reads the whole document plus one chunk, writes a one-line context, and we index context plus chunk in BM25 and in the dense index alike. The cache meter stays at zero because our toy documents are under Haiku 5.5's 512-token cache minimum and the API returns no error; on real policy PDFs you should see cache_write on the first chunk and cache_read after it. Analogy: a librarian writes a short shelf label on each loose page so that it can be found again. Likely question: "Is the LLM call worth it at $0.10 per million input tokens?"
