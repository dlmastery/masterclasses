# lab6_faiss_bench.py
import time
import faiss
import numpy as np
rng = np.random.default_rng(7)
N, D, NQ, K = 100_000, 256, 500, 10
centers = rng.standard_normal((1_000, D)).astype(np.float32)  # clustered, like real embeddings
xb = centers[rng.integers(0, 1_000, N)] + 0.3 * rng.standard_normal((N, D)).astype(np.float32)
xq = xb[rng.choice(N, NQ, replace=False)] + 0.1 * rng.standard_normal((NQ, D)).astype(np.float32)
faiss.normalize_L2(xb); faiss.normalize_L2(xq)                 # cosine == inner product
flat = faiss.IndexFlatIP(D); flat.add(xb)
_, gt = flat.search(xq, K)
recall = lambda I: np.mean([len(set(I[i]) & set(gt[i])) / K for i in range(NQ)])

def bench(index, params, label):
    t = time.perf_counter(); _, I = index.search(xq, K, params=params)
    print(f"{label:<22} recall@10={recall(I):.3f}  {1e3 * (time.perf_counter() - t) / NQ:.3f} ms/q")
hnsw = faiss.IndexHNSWFlat(D, 32, faiss.METRIC_INNER_PRODUCT)  # M=32
hnsw.hnsw.efConstruction = 80
t = time.perf_counter(); hnsw.add(xb); print(f"HNSW build {time.perf_counter() - t:.1f}s")
for ef in (16, 64, 128, 256):
    bench(hnsw, faiss.SearchParametersHNSW(efSearch=ef), f"HNSW efSearch={ef}")

nlist = 1024                                                   # ~4*sqrt(N)
ivf = faiss.IndexIVFFlat(faiss.IndexFlatIP(D), D, nlist, faiss.METRIC_INNER_PRODUCT)
ivf.train(xb[rng.choice(N, 60_000, replace=False)]); ivf.add(xb)
for p in (1, 8, 32, 64):
    bench(ivf, faiss.SearchParametersIVF(nprobe=p), f"IVF nprobe={p}")

Run exact search on a 1-5k query sample. Tune to a recall target (0.95). Write the value into config.
The 41 claims chunks are too few to benchmark, so this step uses 100,000 clustered synthetic vectors at the same 256 dims as the Qwen3 claims vectors. This is the one benchmark every team should run before choosing index settings: exact search gives the true top 10, and each approximate setting is scored against it. Step through HNSW first: the default efSearch of 16 gets only 0.781 recall here, and efSearch 64 crosses the 0.95 target at about a tenth of a millisecond. Then IVF: the default nprobe of 1 finds only 0.412, and you need nprobe 32 to cross the target, at four times the HNSW latency. Think of efSearch and nprobe as how many shelves a librarian checks before answering: more shelves, slower but more complete. The numbers are illustrative; the shape is what you will see on your own data. A likely question is "why not just set efSearch to 256?": because latency grows with it, so pick the smallest value that meets your recall target and write it into config.
