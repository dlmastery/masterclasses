# eval_gate.py -- block a PR when faithfulness drops
import asyncio, json, math, os, sys
from openai import AsyncOpenAI
from ragas.llms import llm_factory
from ragas.metrics.collections import Faithfulness

client = AsyncOpenAI(api_key=os.environ["GEMINI_API_KEY"],
                     base_url="https://generativelanguage.googleapis.com/v1beta/openai/")
faith = Faithfulness(llm=llm_factory("gemini-3.8-flash", provider="openai",
                                     client=client, max_tokens=4096))
base = json.load(open("baseline.json"))                    # {"faithfulness": 0.91}
rows = [json.loads(line) for line in open("runs/candidate.jsonl", encoding="utf-8")]

async def main():
    results = await asyncio.gather(*(faith.ascore(user_input=r["question"], response=r["response"],
                                     retrieved_contexts=r["retrieved_contexts"]) for r in rows))
    vals = [x.value for x in results if not math.isnan(x.value)]  # ragas: a 429 gives NaN
    score = sum(vals) / len(vals)
    print(f"faithfulness={score:.3f} base={base['faithfulness']:.3f}")
    sys.exit(0 if score >= base["faithfulness"] - 0.03 else 1)

asyncio.run(main())

# eval_gate.py (fixed) -- block a PR when faithfulness OR judge coverage drops
import asyncio, json, math, os, sys
from openai import AsyncOpenAI
from ragas.llms import llm_factory
from ragas.metrics.collections import Faithfulness

MIN_ROWS, MAX_NAN_RATE = 50, 0.05
client = AsyncOpenAI(api_key=os.environ["GEMINI_API_KEY"],
                     base_url="https://generativelanguage.googleapis.com/v1beta/openai/")
faith = Faithfulness(llm=llm_factory("gemini-3.8-flash", provider="openai",
                                     client=client, max_tokens=4096))
base = json.load(open("baseline.json"))                    # {"faithfulness": 0.91}
rows = [json.loads(line) for line in open("runs/candidate.jsonl", encoding="utf-8")]
assert len(rows) >= MIN_ROWS, f"only {len(rows)} rows: too few for a reliable gate"

async def main():
    results = await asyncio.gather(*(faith.ascore(user_input=r["question"], response=r["response"],
                                     retrieved_contexts=r["retrieved_contexts"]) for r in rows))
    vals = [x.value for x in results if not math.isnan(x.value)]
    nan_rate = 1 - len(vals) / len(results)
    score = sum(vals) / len(results)                        # release gate: a NaN row counts as 0
    print(f"faithfulness={score:.3f} base={base['faithfulness']:.3f} coverage={1 - nan_rate:.0%}")
    blocked = score < base["faithfulness"] - 0.03 or nan_rate > MAX_NAN_RATE
    sys.exit(1 if blocked else 0)

asyncio.run(main())

Show the code and the first output, and ask the room why a broken judge made the score go up. Failed judge calls come back as NaN, the filter removes them, and the mean divides by the rows that survived, so the more rows fail, the less of the test set the number describes. In this illustrative run the 40 lost rows were the hard ones, so the mean rose from 0.910 to 0.962 and the PR passed. The fix makes coverage part of the gate: compute nan_rate, block above 5%, refuse runs under 50 rows, and for a release decision score a NaN row as 0, which gives 0.641 here. Analogy: a school that sends its weakest students home on exam day gets a better average, not better students. Likely follow-up: "Why not just retry the failed rows?" Retry with backoff, yes, but the gate must still fail closed when coverage is low, and the canary-abstention check runs next to it.
