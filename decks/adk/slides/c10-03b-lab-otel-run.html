async def main() -> None:
    runner = Runner(app=app, session_service=InMemorySessionService(),
                    auto_create_session=True)
    msg = types.Content(role="user",
        parts=[types.Part(text="What is my salary band?")])
    async for event in runner.run_async(
            user_id="E-1001", session_id="s1", new_message=msg,
            state_delta={"employee_id": "E-1001"}):  # guard input
        if event.is_final_response():
            print("agent>", event.content.parts[0].text)
    tp = trace.get_tracer_provider()  # BatchSpanProcessor queue
    if hasattr(tp, "force_flush"):    # ProxyTracerProvider has none
        tp.force_flush()

asyncio.run(main())

state_delta puts the caller's id in state. The guard checks it.

Now we run one turn of the guarded app and read the trace it leaves. Press once: Runner takes the App from step one, so both guard plugins run; state_delta writes the caller's employee id into session state, which is what HrGuardPlugin compares with. Press again: spans leave through a BatchSpanProcessor, so a short script must flush before it exits; the hasattr check is there because with no exporter configured the global provider is a ProxyTracerProvider, which has no force_flush. Press again: the waterfall has four spans, invoke_agent, generate_content, execute_tool and a second generate_content; 2.13 defaults to the GenAI semantic conventions 1.44, which removed the call_llm span, so do not build a dashboard on it. Press again: with TO_CLOUD=1 the same waterfall shows in Cloud Trace. Only adk deploy agent_engine with --otel_to_cloud also turns content capture off; on Cloud Run and GKE you keep the env line from the last slide.
