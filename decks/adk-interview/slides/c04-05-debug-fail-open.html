from google.adk import Agent
from google.adk.tools import FunctionTool, ToolContext

def issue_refund(order_id: str, amount_usd: float) -> dict:
    """Refund to the original payment method. Irreversible."""
    return {"status": "success", "refund_id": f"RF-{order_id}"}

def needs_approval(order_id: str, amount_usd: float,
                   tool_context: ToolContext) -> bool:
    limit = tool_context.state.get("app:approval_limit_usd")
    if limit is None:
        return              # limit not set in this environment
    return amount_usd > float(limit)

root_agent = Agent(name="refund_agent", model="gemini-3.5-flash",
    tools=[FunctionTool(issue_refund,
                        require_confirmation=needs_approval)])
