# probe_base.py - is RFT worth it for claim payouts?
import re
from vllm import LLM, SamplingParams
from claims_data import claims_dataset   # claim prompts + gold "solution"

def is_correct(text, gold):              # the payout must be in \boxed{N}
    m = re.findall(r"\\boxed\{([\d.,]+)\}", text)
    return bool(m) and abs(float(m[-1].replace(",", "")) - float(gold)) < 0.01

N, ds = 8, claims_dataset(200, seed=7)
llm = LLM(model="Qwen/Qwen2.5-0.5B-Instruct")
params = SamplingParams(n=N, temperature=1.0, max_tokens=48)
outs = llm.chat([r["prompt"] for r in ds], params)
solved = [sum(is_correct(o.text, r["solution"]) for o in out.outputs)
          for out, r in zip(outs, ds)]
pass_1 = sum(solved) / (N * len(ds))
pass_n = sum(k > 0 for k in solved) / len(ds)
print(f"pass@1={pass_1:.2f}  pass@{N}={pass_n:.2f}")
if pass_n == 0:
    print("no correct rollout: run SFT first")
elif pass_1 > 0.95:
    print("already solved: RFT is unnecessary")
else:
    print("learning signal exists: run RFT")

params = SamplingParams(n=N, temperature=1.0, max_tokens=384)
outs = llm.chat([r["prompt"] for r in ds], params)
cut = sum(o.finish_reason == "length"
          for out in outs for o in out.outputs) / (N * len(ds))
print(f"truncated={cut:.0%}")                # truncated=3%
assert cut < 0.10, "raise max_tokens before you trust pass@k"
Show the script and its output, and ask: "Do you trust this verdict?" The rule in the script is right: RFT needs at least one correct answer among several attempts, so pass@8 of zero normally means SFT first. The probe is wrong. The grader expects the final payout inside \boxed{}, because the claims prompts ask for that format, the same convention the RL trainer's accuracy reward reads. Click 1, bug 1: 48 tokens is too short for the model to reason and then write the boxed payout, so every one of the 1,600 rollouts is cut off and scored wrong. The 48 probably came from a short-answer default. Set the cap from the length of correct reference answers, for example the 95th percentile plus a margin; here that gives 384. Click 2, bug 2: the script never reads finish_reason, so a cut-off answer looks the same as a wrong answer. Count the "length" stops and trust pass@k only when fewer than 10% are cut. The same signal in GRPO training is completions/clipped_ratio. Click 3: with the fix, truncation drops to 3% and pass@8 rises to 0.64, so the decision flips to RFT. These numbers are illustrative, not a measured run. Analogy: you judge a student on an exam, but you take the paper away after the first line. Likely follow-up: "What else can fake a zero?" A reward parser that never matches, which we will see later in the course, so always unit-test the grader on a real model-style string.
