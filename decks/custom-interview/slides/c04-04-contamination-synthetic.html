ingest.py · datasets 4.8.5
import datetime as dt, hashlib, json, pathlib
from datasets import load_dataset

def ingest(d, source, revision, license_id, generator=None):
    shards = sorted(pathlib.Path(d).glob("*.parquet"))
    rec = {"source": source, "revision": revision,
           "license": license_id, "generator": generator,
           "sha256": {p.name: hashlib.sha256(p.read_bytes())
                      .hexdigest() for p in shards},
           "ingested": dt.date.today().isoformat()}
    pathlib.Path(d, "PROVENANCE.json").write_text(
        json.dumps(rec, indent=2))
    return load_dataset("parquet", split="train",
                        data_files=[str(p) for p in shards])
