import json, re

def grade(sample: dict, item: dict) -> float:
    m = re.search(r"\{.*\}", sample["output_text"], re.S)
    try:
        out = json.loads(m.group(0))
        amount = float(out["refund_amount"])
    except (AttributeError, KeyError, TypeError, ValueError):
        return 0.0                         # no usable JSON
    if out.get("decision") != item["gold_decision"]:
        return 0.0                         # wrong decision
    return 1.0 if abs(amount - item["gold_amount"]) <= 0.01 else 0.5

sample = {"output_text": '{"decision": "approve", "refund_amount": 80}'}
item = {"gold_decision": "approve", "gold_amount": 89.0}   # one JSONL line
print(grade(sample, item))
