# lab4_paired.py - refund agent v1 vs v2 on the SAME 400 golden cases (scipy 1.18.1)
import numpy as np
from scipy import stats

rng = np.random.default_rng(0)
n = 400
difficulty = rng.normal(0, 1.5, n)                     # shared item difficulty
a = (rng.random(n) < 1 / (1 + np.exp(-(1.3 - difficulty)))).astype(float)  # v1
b = (rng.random(n) < 1 / (1 + np.exp(-(1.6 - difficulty)))).astype(float)  # v2
print(f"v1={a.mean():.3f}  v2={b.mean():.3f}  delta={b.mean() - a.mean():+.3f}")

def diff_of_means(x, y, axis=-1):
    return np.mean(x, axis=axis) - np.mean(y, axis=axis)

res = stats.bootstrap((b, a), diff_of_means, paired=False, n_resamples=9999,
                      confidence_level=0.95, method="percentile", rng=rng)
lo, hi = res.confidence_interval
print(f"unpaired 95% CI [{lo:+.3f}, {hi:+.3f}]")
print("ship v2" if lo > 0 else "no evidence: keep v1")

Show the code and the output first, and ask the candidate whether v2 should ship. The bug is on the bootstrap line: paired=False resamples the two models independently, but both answered the same 400 cases, so shared item difficulty stays in the variance and the CI of minus 0.007 to plus 0.110 crosses zero. With paired=True one index vector resamples both arrays, the CI becomes plus 0.005 to plus 0.102, and the exact McNemar test on the 40 versus 61 discordant pairs gives p = 0.046. Analogy: to compare two runners, time them on the same track on the same day; comparing times from different tracks adds the track to the noise. The third trap is operational: when v2 crashed on 3 items and the rows were dropped, SciPy refused the paired call, so score crashes as 0 and align on item_id. Likely follow-up: "How many items do you need?" The minimum detectable effect is about 2.8 times the SE, so 400 paired golden cases detect about 7 points and unpaired about 8.5, the same numbers as the first drill of this chapter.
