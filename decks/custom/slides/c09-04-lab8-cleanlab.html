import numpy as np
from cleanlab.filter import find_label_issues
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import StratifiedKFold, cross_val_predict
from sklearn.pipeline import make_pipeline
from tickets import make_tickets   # classes = sorted(TEMPLATES): 4 support classes

texts, true_y, classes = make_tickets(3_000, seed=1)
# Simulate a rushed BPO labeling vendor: 8% of labels flipped to a random other class.
rng = np.random.default_rng(2)
noisy_y = true_y.copy()
flip = rng.random(len(true_y)) < 0.08
noisy_y[flip] = (true_y[flip] + rng.integers(1, len(classes), flip.sum())) % len(classes)

model = make_pipeline(TfidfVectorizer(ngram_range=(1, 2), min_df=2),
                      LogisticRegression(max_iter=2_000))
# Out-of-sample probabilities: every row is scored by a model that never saw it.
pred_probs = cross_val_predict(model, texts, noisy_y, method="predict_proba",
                               cv=StratifiedKFold(n_splits=5, shuffle=True, random_state=0))

ranked = find_label_issues(labels=noisy_y, pred_probs=pred_probs,
                           return_indices_ranked_by="self_confidence",
                           n_jobs=1)   # Windows/macOS + notebooks: avoid spawn hangs
truly_wrong = set(np.flatnonzero(flip))
hits = len(truly_wrong & set(ranked))
print(f"flipped labels: {len(truly_wrong)}   flagged by cleanlab: {len(ranked)}")
print(f"precision: {hits / len(ranked):.2f}   recall: {hits / len(truly_wrong):.2f}")
print("\nTop 5 for human review:")
for i in ranked[:5]:
    print(f"  #{i:<5} given={classes[noisy_y[i]]:<9} suggested={classes[pred_probs[i].argmax()]:<9}"
          f" p(given)={pred_probs[i, noisy_y[i]]:.2f}  '{texts[i][:48]}'")

This is step 3 of Lab 8: a vendor labeled 3,000 tickets and about 8 percent of the labels are wrong, so we rank the suspects instead of re-reviewing everything. Step 1 flips 8 percent of labels on purpose, which gives 225 known errors; the labels are integers in sorted class order so that column j of pred_probs means class j. Step 2 is the key line: cross_val_predict gives out-of-sample probabilities, because a model scoring its own training rows has memorized the noisy labels and cleanlab then finds almost nothing. Step 3: find_label_issues with return_indices_ranked_by gives 223 indices sorted worst first, with precision 1.00 and recall 0.99, but only because the synthetic templates separate cleanly, so expect much lower numbers on real tickets. Analogy: a spell checker underlines words; it does not delete them, and a person decides. Likely question: how do I make the list more reliable? Run two model families, for example TF-IDF plus LR and a sentence-embedding model, review the intersection first, and log every reviewer decision as fix, keep, ambiguous or drop, because many ambiguous items mean the taxonomy needs work.
