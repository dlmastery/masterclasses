import sys
from peft import LoraConfig
from transformers import TrainerCallback
from trl import GRPOConfig, GRPOTrainer
from trl.rewards import accuracy_reward
from claims_data import claims_dataset
from lab4_rewards import gated_brevity_reward
BETA = float(sys.argv[1]) if len(sys.argv) > 1 else 0.0   # run once with 0.0, once with 0.04
class HackTripwire(TrainerCallback):   # stop when accuracy falls, whatever the total does
    def __init__(self, patience=3):
        self.best_acc, self.bad, self.patience = -1.0, 0, patience
    def on_log(self, args, state, control, logs=None, **kwargs):
        acc = (logs or {}).get("rewards/accuracy_reward/mean")
        if acc is None: return
        if acc > self.best_acc:
            self.best_acc, self.bad = acc, 0
        elif acc < self.best_acc - 0.05:
            self.bad += 1
            print(f"[tripwire] step {state.global_step}: accuracy {acc:.3f} < best {self.best_acc:.3f}")
            control.should_training_stop = self.bad >= self.patience
args = GRPOConfig(
    output_dir=f"claims-kl{BETA}", model_init_kwargs={"dtype": "bfloat16"},
    learning_rate=1e-5, per_device_train_batch_size=8, gradient_accumulation_steps=4,
    num_generations=8, max_completion_length=384,
    beta=BETA,  # > 0 adds KL to the start model and logs `kl` (LoRA: adapter-off = reference)
    reward_weights=[1.0, 0.3],
    multi_objective_aggregation="normalize_then_sum",   # GDPO-style per-reward normalisation
    max_steps=300, logging_steps=5, report_to="trackio", run_name=f"kl{BETA}",
)
trainer = GRPOTrainer(
    model="Qwen/Qwen2.5-0.5B-Instruct",
    reward_funcs=[accuracy_reward, gated_brevity_reward],
    args=args, train_dataset=claims_dataset(2000, seed=7),
    peft_config=LoraConfig(r=16, lora_alpha=32, target_modules="all-linear", task_type="CAUSAL_LM"),
    callbacks=[HackTripwire()],
)
trainer.train()

The last step of the claims lab runs the same job twice, once with beta 0.0 and once with beta 0.04, and adds a tripwire. Step 1: beta above zero adds a KL penalty to the start model, and with LoRA TRL does not load a second model, because the reference is the same weights with the adapter disabled; the kl metric only appears when beta is not zero. The curves are illustrative: without KL the run gains faster and gets shorter, with KL entropy stays higher and length falls more slowly. Step 2: normalize_then_sum z-scores each reward inside its group before the 1.0 and 0.3 weights, so a low-variance brevity reward cannot be drowned out or dominate by scale. Step 3: the tripwire is like a smoke detector that ignores the thermostat; total reward can rise, but if accuracy sits 0.05 under its best for three logs, training stops. Rule of thumb: a verifiable reward starts at beta 0.0, a learned or judge reward keeps beta above zero. Likely question: "Why patience 3?" One bad log can be noise; three bad logs at logging_steps 5 span at least 15 steps of real decline. Both runs use the same seed-7 training set as step 2, so the only change is beta and the brevity reward.
