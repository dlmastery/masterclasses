import numpy as np
from sklearn.datasets import make_classification
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import average_precision_score
from sklearn.model_selection import train_test_split
X, y = make_classification(n_samples=20_000, n_features=30, n_informative=8,
                           weights=[0.9, 0.1], flip_y=0.02, class_sep=0.8, random_state=0)
X_pool, X_test, y_pool, y_test = train_test_split(X, y, test_size=0.25,
                                                  stratify=y, random_state=0)
SEED_SIZE, BATCH, ROUNDS = 100, 100, 8      # adjuster time: 100 claims per day
def margin_scores(model, X_unlabeled):   # small top-2 gap = model is unsure
    p = np.sort(model.predict_proba(X_unlabeled), axis=1)
    return p[:, -1] - p[:, -2]
def run(strategy, seed=0):
    rng = np.random.default_rng(seed)
    labeled = rng.choice(len(X_pool), SEED_SIZE, replace=False)
    curve = []
    for _ in range(ROUNDS + 1):
        model = LogisticRegression(max_iter=1_000, class_weight="balanced")
        model.fit(X_pool[labeled], y_pool[labeled])  # y_pool = the adjuster's answers
        curve.append(average_precision_score(y_test, model.predict_proba(X_test)[:, 1]))
        unlabeled = np.setdiff1d(np.arange(len(X_pool)), labeled)
        if strategy == "margin":
            pick = unlabeled[np.argsort(margin_scores(model, X_pool[unlabeled]))[:BATCH]]
        else:
            pick = rng.choice(unlabeled, BATCH, replace=False)
        labeled = np.concatenate([labeled, pick])
    return np.array(curve)
rand = np.mean([run("random", s) for s in range(5)], axis=0)
marg = np.mean([run("margin", s) for s in range(5)], axis=0)
print(f"{'labels':>7} {'random PR-AUC':>14} {'margin PR-AUC':>14}")
for r, (a, b) in enumerate(zip(rand, marg)):
    print(f"{SEED_SIZE + r * BATCH:>7} {a:>14.3f} {b:>14.3f}")
Seed: 100 random claims, about 10 positive. class_weight="balanced" stops an all-"auto-approve" model.
Margin = p(top-1) − p(top-2). Smallest margin = least sure. The same code works for K classes.
Retrain from scratch each round. Score on a fixed test set that never enters the pool.
Same budget, random picks: PR-AUC 0.548 vs 0.599 at 200 labels. Next: the full curve.

This is step 3 of Lab 6: senior adjusters label only 100 claims per day, so we let the model choose which 100. Step 1: the seed is 100 random claims, which hold only about 10 positives, so class_weight="balanced" is what stops the first model from predicting auto-approve for everything. Step 2: the margin is the gap between the top two class probabilities; a small gap means the model is unsure, and the same code works for K classes. Step 3: each round we retrain from scratch and score PR-AUC on a fixed test set that never enters the pool, because queried items are hard by construction and would bias the estimate. Step 4: random picks with the same budget land mostly far from the boundary, and at 200 labels random scores 0.548 while margin scores 0.599. Analogy: a teacher who reviews only the exam answers where the student hesitated.
