import numpy as np
from sklearn.calibration import CalibratedClassifierCV, calibration_curve
from sklearn.datasets import make_classification
from sklearn.frozen import FrozenEstimator
from sklearn.metrics import brier_score_loss, roc_auc_score
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB

def ece(y, p, n_bins=15, strategy="uniform"):       # strategy: 'uniform' or 'quantile'
    if strategy == "uniform":
        edges = np.linspace(0, 1, n_bins + 1)
    else:
        edges = np.quantile(p, np.linspace(0, 1, n_bins + 1))
    idx = np.clip(np.searchsorted(edges, p, side="right") - 1, 0, n_bins - 1)
    total = 0.0
    for b in range(n_bins):
        m = idx == b
        if m.any():
            total += m.mean() * abs(y[m].mean() - p[m].mean())
    return total

X, y = make_classification(n_samples=20_000, n_features=20, n_informative=6,
                           n_redundant=10, weights=[0.9, 0.1], random_state=0)
X_tr, X_rest, y_tr, y_rest = train_test_split(X, y, test_size=0.5, stratify=y, random_state=0)
X_cal, X_te, y_cal, y_te = train_test_split(X_rest, y_rest, test_size=0.5, stratify=y_rest, random_state=0)
base = GaussianNB().fit(X_tr, y_tr)        # NB with redundant features = overconfident
models = {"raw GaussianNB": base}
for method in ["sigmoid", "isotonic", "temperature"]:
    models[method] = CalibratedClassifierCV(FrozenEstimator(base), method=method).fit(X_cal, y_cal)
print(f"{'model':15s} {'ROC-AUC':>7s} {'Brier':>7s} {'ECE-15u':>8s} {'ECE-15q':>8s}")
for name, m in models.items():
    p = m.predict_proba(X_te)[:, 1]
    print(f"{name:15s} {roc_auc_score(y_te, p):7.3f} {brier_score_loss(y_te, p):7.4f} "
          f"{ece(y_te, p):8.4f} {ece(y_te, p, strategy='quantile'):8.4f}")
frac_pos, mean_pred = calibration_curve(y_te, base.predict_proba(X_te)[:, 1], n_bins=10)
print("raw reliability (mean_pred -> frac_pos):",
      [f"{mp:.2f}->{fp:.2f}" for mp, fp in zip(mean_pred, frac_pos)])

Step 1: the reliability curve of the raw Naive Bayes model shows that when it says 0.86, only 35% of those refunds are fraud, because Naive Bayes counts the ten redundant features many times. Step 2: we wrap the fitted model in FrozenEstimator and fit only a calibrator on a separate calibration split, so sigmoid cuts ECE from 0.0677 to 0.0173 and isotonic to 0.0063. Step 4: ROC-AUC is identical in every row, because a monotone map changes the numbers, not the order; temperature scaling made ECE worse, 0.0780, because it has one parameter and no intercept, and the sklearn docstring says sigmoid "might be preferred" for very uncalibrated models on imbalanced data.
