def run_tests(code: str, tests: str, timeout: int = 5) -> tuple[bool, str]:
    with tempfile.TemporaryDirectory() as tmp:          # fresh dir per candidate
        path = os.path.join(tmp, "candidate.py")
        with open(path, "w", encoding="utf-8") as f:
            f.write(code + "\n\n" + tests + "\n")
        try:
            r = subprocess.run([sys.executable, "-I", path], cwd=tmp, capture_output=True,
                               text=True, timeout=timeout)
        except subprocess.TimeoutExpired:
            return False, "timeout"
    return r.returncode == 0, r.stderr.strip().splitlines()[-1] if r.stderr.strip() else ""

@step(inputs=["generation", "tests"], outputs=["code", "passed", "error"])
def ExecuteUnitTests(inputs: StepInput) -> StepOutput:
    for row in inputs:
        m = re.search(r"```(?:python)?\n(.*?)```", row["generation"] or "", re.S)
        row["code"] = m.group(1) if m else (row["generation"] or "")
        row["passed"], row["error"] = run_tests(row["code"], row["tests"])
    yield inputs

with Pipeline(name="hr-tools-exec-verify") as pipeline:
    seeds = LoadDataFromDicts(data=SEEDS)
    gen = TextGeneration(
        llm=OpenAILLM(model=os.getenv("GEN_MODEL", "gpt-4.1-mini"),
                      generation_kwargs={"temperature": 0.9, "max_new_tokens": 512}),
        template=TEMPLATE, columns=["spec"], num_generations=4)
    verify = ExecuteUnitTests()
    seeds >> gen >> verify

if __name__ == "__main__":
    ds = pipeline.run(use_cache=False)["default"]["train"]
    sft = ds.filter(lambda r: r["passed"])
    print(f"candidates={len(ds)}  passed={len(sft)}  pass-rate={len(sft) / len(ds):.0%}")
    for r in ds.filter(lambda r: not r["passed"]).select_columns(["spec", "error"]).to_list()[:3]:
        print("REJECTED:", r["spec"].splitlines()[0], "->", r["error"])
    sft.select_columns(["spec", "code"]).to_json("hr_tools_sft.jsonl")

Lab 9 verified a synthetic eval set; now we add a second kind of synthetic data: SFT rows for the Python tools that the HR assistant calls, pto_days (1.5 days per month of service) and expense_on_time (30-day filing deadline), each with 4 hidden asserts. The before version writes whatever the LLM produced straight into the training file, so not one row was ever run. The after version gives each spec 4 candidates and runs each one in a fresh temp directory with python -I and a 5 second timeout; the verifier itself was re-run and returned pass, AssertionError and timeout as expected. The error column becomes a rejection histogram, and the passing and failing candidates for one spec give you DPO pairs for free. Say this clearly: distilabel's last release is 1.5.3 from January 2025 and the README says the original authors moved on, so teach the pattern and keep the verifier a plain function you can move to any framework. Analogy: a code review that only accepts pull requests with green CI. Likely question: "Why not let an LLM judge the code?" Judges cost more and are less reliable than tests, so put them last, only on rows that already pass execution.
