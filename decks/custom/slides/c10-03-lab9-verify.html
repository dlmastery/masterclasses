# lines 1-20 (step 1): import collections, difflib, hashlib, json; client, JUDGE, rows
def ask_judge(prompt: str) -> str:
    out = client.chat.completions.create(model=JUDGE, temperature=0, messages=[
        {"role": "user", "content": prompt + '\nReply as JSON: {"verdict": "YES" or "NO"}'}],
        response_format={"type": "json_object"})
    return json.loads(out.choices[0].message.content)["verdict"]

kept, dropped = [], collections.Counter()
for r in rows:
    ctx = "\n---\n".join(r["reference_contexts"])
    if ask_judge(f"Context:\n{ctx}\n\nIs every fact in this answer stated in the context?\n"
                 f"Answer: {r['reference']}") != "YES":
        dropped["reference not grounded"] += 1; continue
    closed_book = client.chat.completions.create(model=JUDGE, temperature=0, messages=[
        {"role": "user", "content": r["user_input"]}]).choices[0].message.content
    if ask_judge(f"Reference: {r['reference']}\nCandidate: {closed_book}\n"
                 "Does the candidate state the same specific facts as the reference?") == "YES":
        dropped["answerable without retrieval"] += 1; continue
    if any(difflib.SequenceMatcher(None, r["user_input"].lower(), k["user_input"].lower()).ratio() > 0.9
           for k in kept):
        dropped["near-duplicate question"] += 1; continue
    kept.append(r)

blob = "\n".join(json.dumps(r, sort_keys=True) for r in kept)
version = hashlib.sha256(blob.encode()).hexdigest()[:12]
with open(f"hr_rag_eval_{version}.jsonl", "w", encoding="utf-8") as f:
    f.write(blob + "\n")
print(f"generated={len(rows)} kept={len(kept)} dropped={dict(dropped)}")
print("synthesizer mix:", collections.Counter(r["synthesizer_name"] for r in kept))
print(f"wrote hr_rag_eval_{version}.jsonl  -> send 10% to an HR reviewer before first use")

Gate 1: every fact in the reference must be in its contexts. Keep reference_contexts and score retrieval recall on its own.
Gate 2: the judge answers with no contexts. The same facts mean parametric memory, not RAG. Drop the item or tag its own slice.
Gate 3: a difflib ratio above 0.9 against a kept question drops the twin.

Step 1 of the lab gave us 40 synthetic question, answer and context rows from ragas; this step treats them as a draft and lets only verified rows into the eval set. First the judge helper: temperature zero and JSON mode, so every check returns one YES or NO that the code can parse. Gate 1 drops rows whose reference answer states facts that are not in its own contexts; gate 2 asks the judge the question with no contexts at all, and if it already knows the answer, the row tests the model's memory, not our retrieval; gate 3 drops near-identical questions with a difflib ratio above 0.9. The 29 survivors are written as sorted JSON and named by their sha256, so nobody can change the set without changing its name, and the quality card in chapter 11 cites that hash. Analogy: an exam board that throws out questions a student could answer without the textbook. Likely question: "Is the judge itself trustworthy here?" Not yet: that is why 10% goes to an HR reviewer, and step 4 of the lab measures the judge against human labels. Next, step 3 protects this frozen set from the other side: no training text may copy it.
