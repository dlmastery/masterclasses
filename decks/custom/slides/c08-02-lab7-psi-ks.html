import numpy as np
from scipy.stats import ks_2samp

def psi(reference, current, n_bins=10, eps=1e-4):
    """Population Stability Index with quantile bins fixed on the REFERENCE window."""
    edges = np.quantile(reference, np.linspace(0, 1, n_bins + 1))
    edges[0], edges[-1] = -np.inf, np.inf          # catch out-of-range current values
    edges = np.unique(edges)                       # ties (e.g. many zeros) collapse bins
    ref_share = np.histogram(reference, edges)[0] / len(reference)
    cur_share = np.histogram(current, edges)[0] / len(current)
    ref_share, cur_share = np.clip(ref_share, eps, None), np.clip(cur_share, eps, None)
    return float(np.sum((cur_share - ref_share) * np.log(cur_share / ref_share)))

def verdict(value):
    return "stable" if value < 0.1 else ("moderate" if value < 0.25 else "MAJOR")

rng = np.random.default_rng(7)
# Reference: September refund requests. Current: October, after a holiday promo.
reference = {
    "refund_amount_usd": rng.lognormal(mean=3.6, sigma=0.6, size=20_000),
    "days_since_order":  rng.gamma(shape=2.0, scale=6.0, size=20_000),
    "items_in_order":    rng.poisson(lam=2.0, size=20_000).astype(float),
}
current = {
    "refund_amount_usd": rng.lognormal(mean=3.9, sigma=0.7, size=5_000),   # real shift
    "days_since_order":  rng.gamma(shape=2.0, scale=6.2, size=5_000),      # tiny shift
    "items_in_order":    rng.poisson(lam=2.0, size=5_000).astype(float),   # no shift
}

print(f"{'feature':<20}{'PSI':>8}  {'verdict':<9}{'KS D':>7}{'KS p':>11}")
for col in reference:
    p = psi(reference[col], current[col])
    ks = ks_2samp(reference[col], current[col])
    print(f"{col:<20}{p:>8.3f}  {verdict(p):<9}{ks.statistic:>7.3f}{ks.pvalue:>11.2e}")

Lab 7 starts with drift on the Acme refund-fraud model: 20,000 September refund requests are the reference and 5,000 October requests, after the holiday promo, are the current window. Step 1: the bins are September deciles and stay fixed, so every blue bar holds 10% and next week's PSI is comparable with this week's; step 2: the plus and minus infinity edges catch the October refunds above September's maximum, and that open top bin holds 26% of October. Step 3: np.unique collapses the tied edges of the count feature to 6 bins, np.clip keeps an empty bin from giving log of zero, and the numbers come out: refund amount PSI 0.243, moderate. Step 4 is the lesson: days_since_order has PSI 0.003, stable, but the KS p-value is 0.02, so a p-value test at alpha 0.05 calls a tiny shift "drift" because n is large; this is why Evidently switches to distance metrics above 1,000 rows. Analogy: with a big enough crowd, a hearing test detects a whisper that nobody cares about. Likely question: "Are 0.1 and 0.25 standard?" No, they are a credit-risk convention, so set your own bands by comparing two halves of September with each other. Hand-off: drift tells us the data moved; step 2 asks whether the validation score can be trusted at all.
