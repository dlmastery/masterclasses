"""Lab 10 step 2 - claims agent v1 vs v2 on the SAME 400 golden cases (simulated)."""
import numpy as np
from scipy import stats
from statsmodels.stats.contingency_tables import mcnemar

rng = np.random.default_rng(0)
n = 400
difficulty = rng.normal(0, 1.5, n)                     # shared item difficulty
a = (rng.random(n) < 1 / (1 + np.exp(-(1.3 - difficulty)))).astype(float)  # v1
b = (rng.random(n) < 1 / (1 + np.exp(-(1.6 - difficulty)))).astype(float)  # v2
print(f"v1={a.mean():.3f}  v2={b.mean():.3f}  delta={b.mean() - a.mean():+.3f}")

def diff_of_means(x, y, axis=-1):
    return np.mean(x, axis=axis) - np.mean(y, axis=axis)

common = dict(n_resamples=9999, confidence_level=0.95, method="percentile", rng=rng)
unpaired = stats.bootstrap((b, a), diff_of_means, paired=False, **common)
paired = stats.bootstrap((b, a), diff_of_means, paired=True, **common)
bca = stats.bootstrap((b - a,), np.mean, method="BCa", n_resamples=9999, rng=rng)
for name, r in [("unpaired", unpaired), ("paired", paired), ("paired BCa", bca)]:
    lo, hi = r.confidence_interval
    print(f"{name:11s} 95% CI [{lo:+.3f}, {hi:+.3f}]  width={hi - lo:.3f}  SE={r.standard_error:.4f}")

# McNemar uses only the discordant pairs (v1 right/v2 wrong vs v1 wrong/v2 right).
table = [[np.sum((a == 1) & (b == 1)), np.sum((a == 1) & (b == 0))],
         [np.sum((a == 0) & (b == 1)), np.sum((a == 0) & (b == 0))]]
res = mcnemar(table, exact=True)
print(f"discordant: v1-only={table[0][1]} v2-only={table[1][0]}  McNemar exact p={res.pvalue:.4f}")

# Minimum detectable effect (alpha=0.05, power=0.8) ~ 2.8 * SE(delta)
for name, r in [("unpaired", unpaired), ("paired", paired)]:
    print(f"MDE {name:8s} ~ {2.8 * r.standard_error:.3f}")

Step 1 of the lab gave us slice CIs for v2; now the question is whether v2 of the claims agent is really better than v1 on the same 400 golden cases. This is a second file: the golden set, with one pass or fail per case for each version, not the 353-question slice log from step 1. Both arrays come from one shared difficulty per case, so a hard case tends to fail for both versions; that is why the tiles in the grid cluster into both-right and both-wrong. With the same data, the unpaired bootstrap gives a CI that crosses zero, while the paired bootstrap, which resamples one index vector for both arrays, stays above zero at plus 0.5 to plus 10 points. McNemar makes the same point from the other side: the 299 concordant cases carry no information about the difference, only the 40 versus 61 discordant cases do, and the exact p is 0.046. Analogy: to compare two runners, time them on the same track on the same day instead of on two random tracks. Likely question: "Can I skip the MDE?" No: at 400 paired cases you can detect only about 7 points, so a 3-point improvement needs a bigger eval set before anyone builds it. Next step: write these numbers into the quality card.
