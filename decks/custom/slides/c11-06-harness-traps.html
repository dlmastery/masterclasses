score_fix.py · plain Python 3.12, no harness API
import re

LETTER = re.compile(r"^\s*(?i:answer\s*[:=]\s*)?\(?([A-D])\b")

def extract_letter(output: str) -> str | None:          # 'B. 4', 'ANSWER: B' -> 'B'
    m = LETTER.match(output)
    return m.group(1) if m else None

def score_all(outputs: list[str], golds: list[str]) -> dict:
    preds = [extract_letter(o) for o in outputs]
    scores = [1.0 if p == g else 0.0 for p, g in zip(preds, golds, strict=True)]
    errored = sum(p is None for p in preds)              # refusal / unparseable -> 0.0
    return {"accuracy": sum(scores) / len(scores),        # denominator = ALL items
            "scored": len(scores) - errored, "errored": errored, "unscored": 0}

def cache_path(model: str, sha: str) -> str:              # lm_eval --use_cache PATH
    return f"cache/{model}@{sha}"                          # one cache per model@revision

print(score_all(["B", "B. 4", "ANSWER: B", "I cannot answer."], ["B"] * 4))
print(cache_path("claims-agent-v2", "4f1c9e2"))
