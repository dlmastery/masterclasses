import sys
import numpy as np
import pandas as pd
from evidently import DataDefinition, Dataset, Report
from evidently.metrics import ValueDrift
from evidently.presets import DataDriftPreset

rng = np.random.default_rng(42)
def make_window(n, outage_share):   # one synthetic window of tickets
    p_area = [0.35, 0.30 - outage_share / 2, outage_share, 0.35 - outage_share / 2]
    return pd.DataFrame({
        "channel": rng.choice(["email", "chat", "phone"], size=n, p=[0.5, 0.4, 0.1]),
        "product_area": rng.choice(["billing", "login", "outage", "shipping"], size=n, p=p_area),
        "body_chars": rng.lognormal(5.5, 0.5, n).round(),
        "customer_tenure_days": rng.gamma(2.0, 200.0, n).round(),
    })
reference = make_window(4_000, outage_share=0.05)   # last month
current = make_window(1_500, outage_share=0.25)     # this week: login outage
schema = DataDefinition(numerical_columns=["body_chars", "customer_tenure_days"],
                        categorical_columns=["channel", "product_area"])
ref_ds = Dataset.from_pandas(reference, data_definition=schema)
cur_ds = Dataset.from_pandas(current, data_definition=schema)

report = Report([
    DataDriftPreset(num_method="psi", num_threshold=0.2, drift_share=0.5),
    ValueDrift(column="customer_tenure_days", method="ks", threshold=0.05),  # p-value test
], include_tests=True)
snapshot = report.run(current_data=cur_ds, reference_data=ref_ds)
snapshot.save_html("ticket_drift_report.html")

result = snapshot.dict()
for m in result["metrics"]:
    print(f"{m['metric_name'][:60]:<62} {m['value']}")
failed = [t for t in result["tests"] if t["status"] == "FAIL"]
print(f"tests: {len(result['tests'])}, failed: {len(failed)}")
for t in failed:
    print("FAIL:", t["name"], "-", t["description"][:90])
sys.exit(1 if failed else 0)   # non-zero exit blocks the retraining job in CI

Step 1 of Lab 8: a weekly CI job compares 1,500 tickets from this week with 4,000 from last month, writes an HTML report, and exits non-zero when a drift test fails. The 0.7.x imports come from the top-level evidently package; old tutorials that import evidently.report fail, because that API now lives under evidently.legacy. include_tests=True makes it a gate: the Report default is False, so without it there are no tests and the job never fails. run() returns a Snapshot, and snapshot.dict() holds metrics and tests; parse metric_name or id, because there is no metric_id key.
