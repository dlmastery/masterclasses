import datetime as dt, hashlib, json, sys
import pandas as pd
from statsmodels.stats.proportion import proportion_confint
EVAL_FILE = "claims_eval.jsonl"                       # written by step 1
OTHER = {"ece_15_uniform": 0.0063, "calibration": "isotonic on 5,000-item cal split",  # C08 Lab 7
         "threshold": 0.79, "cost_model": "(ok, triage, wrong) = (0, 1, 5)",           # C09 Lab 8
         "judge_kappa": 0.580, "judge_tpr": 0.946, "judge_tnr": 0.593}                  # C10 Lab 9
GATES = {"min_slice_ci_low": 0.50, "min_slice_n": 30, "min_judge_kappa": 0.60}
raw = open(EVAL_FILE, "rb").read()
df = pd.read_json(EVAL_FILE, lines=True)
k, n = int(df.passed.sum()), len(df)
lo, hi = proportion_confint(k, n, method="wilson")
g = df.groupby(["line", "lang"])["passed"].agg(k="sum", n="count").reset_index()
g["lo"], g["hi"] = proportion_confint(g.k, g.n, method="wilson")
failures = [f"slice {r.line}/{r.lang}: CI low {r.lo:.2f} < {GATES['min_slice_ci_low']}"
            for r in g.itertuples() if r.lo < GATES["min_slice_ci_low"]]
failures += [f"judge kappa {OTHER['judge_kappa']:.2f} < {GATES['min_judge_kappa']}"] \
    if OTHER["judge_kappa"] < GATES["min_judge_kappa"] else []
thin = [f"{r.line}/{r.lang} (n={r.n})" for r in g.itertuples() if r.n < GATES["min_slice_n"]]
card = {
    "card_version": "1.0", "created": dt.date.today().isoformat(),
    "eval_set": {"file": EVAL_FILE, "sha256": hashlib.sha256(raw).hexdigest()[:16],
                 "n_items": n, "n_clusters": int(df.claim_id.nunique()), "unit_of_resampling": "claim_id"},
    "headline": {"pass_rate": round(k / n, 3), "wilson_95": [round(lo, 3), round(hi, 3)]},
    "slices": g.round(3).to_dict(orient="records"),
    "calibration": {k_: OTHER[k_] for k_ in ("ece_15_uniform", "calibration")},
    "decision_policy": {k_: OTHER[k_] for k_ in ("threshold", "cost_model")},
    "judge": {k_: OTHER[k_] for k_ in ("judge_kappa", "judge_tpr", "judge_tnr")},
    "known_gaps": {"thin_slices_n_lt_30": thin},
    "gate": {"passed": not failures, "failures": failures},
}
json.dump(card, open("quality_card.json", "w"), indent=2)
md = [f"# Quality card - claims agent eval ({card['created']})",
      f"**Pass rate {k / n:.1%}** (Wilson 95% {lo:.1%}-{hi:.1%})"]
# (8 lines omitted: eval-set line, slice table, judge row, thin slices; same f-string pattern)
md += [f"**Gate: {'PASS' if not failures else 'FAIL'}** " + "; ".join(failures)]
open("quality_card.md", "w").write("\n".join(md) + "\n")
print("\n".join(md))
sys.exit(0 if not failures else 1)

This step turns the numbers from the earlier labs into one artifact that a reviewer reads in a minute and CI can enforce. First, hash the stored bytes of the eval file: if you hash a re-serialized DataFrame, key order and float formatting change the hash on every run, and numbers from two set versions get mixed silently. Second, the GATES dict is written before anyone sees results, and it tests the lower bound of each Wilson CI, so a thin slice with 8 cases cannot pass by luck on a good point estimate. Third, the card records the resampling unit, claim_id, and lists its known gaps, the five slices under n equals 30; an honest card names its blind spots. Analogy: the card is a lab report with the sample ID on it, not a slide with one big number. Likely question: "Why hard-code OTHER?" Only for clarity here; in production every lab writes a JSON fragment and the card script merges them, so nobody retypes a number. The paired v1-vs-v2 result from step 2 goes on the card the same way, as one more fragment. Next step: run the gate.
