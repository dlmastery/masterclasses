import os, re, subprocess, sys, tempfile
CODE_RE = re.compile(r"```python\n(.*?)```", re.DOTALL)
HARNESS = open("harness.tpl").read()  # {code} + the hidden tests; prints the pass rate
def unit_test_reward(completions, tests, log_metric=None, **kwargs):
    rewards, timeouts = [], 0
    for c, t in zip(completions, tests, strict=True):
        m = CODE_RE.search(c[0]["content"])
        if not m:
            rewards.append(0.0)
            continue
        with tempfile.TemporaryDirectory() as d:
            path = os.path.join(d, "cand.py")
            with open(path, "w", encoding="utf-8") as f:
                f.write(HARNESS.format(code=m.group(1)))
            try:
                r = subprocess.run([sys.executable, "-I", path, t], cwd=d,
                                   capture_output=True, text=True, timeout=5)
                rewards.append(float(r.stdout.strip().splitlines()[-1])
                               if r.returncode == 0 else 0.0)
            except subprocess.TimeoutExpired:
                timeouts += 1
                rewards.append(0.0)
            except (ValueError, IndexError):
                rewards.append(0.0)
    if log_metric is not None:
        log_metric("code/timeout_rate", timeouts / len(completions))
    return rewards
