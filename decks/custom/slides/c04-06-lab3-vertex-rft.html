import os
from google import genai
from google.genai import types
PROJECT, LOC = os.environ["GOOGLE_CLOUD_PROJECT"], "us-central1"
client = genai.Client(vertexai=True, project=PROJECT, location=LOC)   # Vertex -> v1beta1
EVALUATE_SRC = r'''
import json, re
def evaluate(example, response) -> float:
    text = "".join(p.get("text", "") for p in response.get("parts", []))
    refs = example.get("references", {})
    m = re.search(r"\{.*\}", text, re.DOTALL)
    if not m:
        return -1.0                  # unparseable = clearly bad, never crash/NaN
    try:
        out = json.loads(m.group(0))
    except Exception:
        return -1.0
    if not {"decision", "refund_amount", "reason"} <= out.keys(): return -1.0  # schema gate
    if out.get("decision") != refs.get("gold_decision"):
        return -0.5
    try:
        ok = abs(float(out.get("refund_amount", -1)) - float(refs.get("gold_amount"))) <= 0.01
    except (TypeError, ValueError):
        ok = False
    return 1.0 if ok else 0.25
'''
reward = types.SingleReinforcementTuningRewardConfig(
    reward_name="refund_policy",
    code_execution_reward_scorer=types.ReinforcementTuningCodeExecutionRewardScorer(
        python_code_snippet=EVALUATE_SRC))
example = types.ReinforcementTuningExample(
    contents=[types.Content(role="user", parts=[types.Part(text=
        "Order A-1182, $49.99, unopened, returned on day 12. "
        "Policy RP-30: full refund within 30 days. "
        'Reply with JSON {"decision", "refund_amount", "reason"}.')])],
    references={"gold_decision": "approve", "gold_amount": "49.99"})
sample = types.Content(role="model", parts=[types.Part(text=
    '{"decision": "approve", "refund_amount": 49.99, "reason": "RP-30, day 12"}')])
check = client.tunings.validate_reward(parent=f"projects/{PROJECT}/locations/{LOC}",
                                       sample_response=sample, example=example,
                                       single_reward_config=reward)
print("overall_reward:", check.overall_reward, "| details:", check.reward_info_details)

This is step 4, the last stop of Lab 3: the same refund logic and the same schema gate as a Vertex code-execution reward, evaluate(example, response), which reads the gold labels from example["references"]. Only the range changes, because Vertex clips to [-1, 1]: -1.0 for unparseable or schema-less output, -0.5 for a wrong decision, 0.25 for the right decision with a wrong amount and 1.0 for a correct answer. The step 2 completions give good 1.0, wrong_amt 0.25, hack -1.0 and broken -1.0 on the CPU, and the regex finds the JSON with or without the answer tags. The key habit is to return a low value instead of raising: if more than 80% of reward invocations error, Vertex stops the job, so validate_reward is your free CI check before you pay for training. Treat it like a smoke test on a staging server: one sample, real sandbox, no rollout cost. The optional tune() call uses batch_size 32 and samples_per_prompt 16, so 512 rollouts per step, which is divisible by 16 and under the 8,192 cap; the feature is Pre-GA, so do not use production or confidential data, and keep the SDK's v1beta1 default. A likely question: "Which metric tells me the reward is too easy or too hard?" Watch /learnable_prompt_ratio, the Vertex analogue of 1 minus TRL's frac_reward_zero_std.
