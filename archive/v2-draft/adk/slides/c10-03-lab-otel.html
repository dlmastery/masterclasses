import asyncio
import os
from opentelemetry import trace
from google.adk import Runner
from google.adk.sessions import InMemorySessionService
from google.adk.telemetry import google_cloud
from google.adk.telemetry.setup import maybe_set_otel_providers
from google.genai import types

os.environ.setdefault("OTEL_SERVICE_NAME", "contoso-hr")
os.environ.setdefault("OTEL_RESOURCE_ATTRIBUTES", "deployment.environment=dev")
os.environ.setdefault("ADK_CAPTURE_MESSAGE_CONTENT_IN_SPANS", "false")  # no PII

hooks = ([google_cloud.get_gcp_exporters(enable_cloud_tracing=True)]
         if os.getenv("TO_CLOUD") else [])
maybe_set_otel_providers(otel_hooks_to_setup=hooks)  # + OTEL_EXPORTER_OTLP_* env

from hr_agent.agent import app  # step 1: HrGuardPlugin + ModelArmorPlugin

async def main() -> None:
    runner = Runner(app=app, session_service=InMemorySessionService(),
                    auto_create_session=True)
    msg = types.Content(role="user", parts=[types.Part(text="What is my salary band?")])
    async for event in runner.run_async(
            user_id="E-1001", session_id="s1", new_message=msg,
            state_delta={"employee_id": "E-1001"}):  # the guard reads it
        if event.is_final_response():
            print("agent>", event.content.parts[0].text)
    tp = trace.get_tracer_provider()  # BatchSpanProcessor: flush before exit
    if hasattr(tp, "force_flush"):    # the default ProxyTracerProvider has none
        tp.force_flush()

asyncio.run(main())

This is step two of the Contoso HR lab: we trace the same guarded app from step one, and we keep employee data out of the spans. The three setdefault lines run before the agent import; the important one is ADK_CAPTURE_MESSAGE_CONTENT_IN_SPANS, which defaults to on, so prompts and replies would land in every span. With no hooks the script exports over OTLP to any collector on port 4318, for example Jaeger; with TO_CLOUD=1 it sends OTLP over HTTP to telemetry.googleapis.com, which needs roles/telemetry.writer. Think of a trace as a flight log that many people can read: it records times and fuel, not the passengers' conversations; the full content belongs in an access-controlled store. A likely question is "why is call_llm missing from my dashboard?": 2.13 defaults to the GenAI semconv 1.44 schema, which removes that span, so build on generate_content instead.
