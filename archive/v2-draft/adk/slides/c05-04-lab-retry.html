# payouts/app.py   (run with: python -m payouts.app start|resume ...)
import asyncio, json, sys
from google.adk import Context, Event, Runner, Workflow
from google.adk.apps import App, ResumabilityConfig
from google.adk.events import RequestInput
from google.adk.sessions.sqlite_session_service import SqliteSessionService
from google.adk.workflow import RetryConfig, node
from google.genai import types
def kyc_lookup(node_input: str) -> dict:
  open("side_effects.log", "a").write(f"kyc_lookup {node_input}\n")  # step 4 counts it
  return {"payee": node_input, "risk": "medium"}
def analyst_review(node_input: dict):        # plain function: a hand-off pause
  yield RequestInput(message=f"Pay {node_input['payee']}? yes/no")
@node(retry_config=RetryConfig(max_attempts=4, initial_delay=0.5,
                               backoff_factor=2.0, jitter=0.0,
                               exceptions=[ConnectionError, "NodeTimeoutError"]),
      timeout=2.0)
async def pay(ctx: Context, node_input: str):  # node_input = the analyst's reply
  yield Event(message=f"gateway attempt {ctx.attempt_count}")
  if node_input.strip().lower() != "yes":
    yield Event(output="CANCELLED"); return
  if ctx.attempt_count < 3:
    raise ConnectionError("gateway 503")       # attempts 1-2 fail, 3 succeeds
  yield Event(output="PAID idempotency_key=payout-" + ctx.invocation_id)
root = Workflow(name="payouts", edges=[("START", kyc_lookup, analyst_review, pay)])
app = App(name="payouts", root_agent=root,
          resumability_config=ResumabilityConfig(is_resumable=True))
runner = Runner(app=app, session_service=SqliteSessionService("payouts.db"))

Attempts 1 and 2 raise ConnectionError. Attempt 3 pays.
exceptions matches class names. Add "NodeTimeoutError" to retry a timeout.
analyst_review pauses so step 4 has a run to resume. Its reply "yes" becomes the node_input of pay.

This is the payout half of the Acme scenario, and it needs no API key because there is no LLM node. Walk the RetryConfig line first: max_attempts counts the first try, so 4 means one try and up to three retries, and with jitter at zero the waits are exactly 0.5 seconds and then 1.0 second, because the delay is initial_delay times backoff_factor to the power attempt minus one. The pay node fails twice with a gateway 503 and pays on attempt 3, which you can see in ctx.attempt_count. The trap is the exceptions list: it matches class names, and a timeout raises NodeTimeoutError, which is not a ConnectionError, so without the string a 2-second timeout is not retried at all. If you write RetryConfig() with no arguments, the defaults are max_attempts=5, initial_delay=1.0, max_delay=60, backoff_factor=2.0 and jitter=1.0, so the wait before try 2 lands anywhere between 0 and 2 seconds; keep jitter on in production to spread load and turn it off in tests. analyst_review is a plain function, so ADK wraps it as a function node with the default hand-off pattern: the reply becomes its output, and that output is the node_input of pay. The output pane needs main(), which is step 4. Think of jitter as staggering the start times of runners so they do not all hit the gateway at the same second. A likely question: "is the attempt counter saved?" No, after an interrupt or a restart the node gets its full retry budget back.
