refund_guard.py
from typing import Any, Optional
from google.adk.plugins.base_plugin import BasePlugin
from google.adk.tools.base_tool import BaseTool
from google.adk.tools.tool_context import ToolContext

class RefundGuardPlugin(BasePlugin):
    def __init__(self, max_amount: float = 500.0):
        super().__init__(name="refund_guard")
        self.max_amount = max_amount

    async def before_tool_callback(
        self, *, tool: BaseTool, tool_args: dict[str, Any], tool_context: ToolContext
    ) -> Optional[dict]:
        amount = float(tool_args.get("amount", 0) or 0)
        if amount > self.max_amount:     # a returned dict skips the tool
            return {"status": "blocked",
                    "reason": f"{amount} exceeds limit {self.max_amount}"}
        return None
guarded_app.py
import asyncio
from google.adk.agents import LlmAgent
from google.adk.apps import App
from google.adk.runners import InMemoryRunner
from refund_guard import RefundGuardPlugin
def issue_refund(order_id: str, amount: float) -> dict:
    """Refund amount (USD) to the card on file for order_id."""
    return {"status": "ok", "refunded": amount}
M, T = "gemini-3.5-flash", [issue_refund]
partner = LlmAgent(name="partner_refunds", model=M, tools=T, description="Partner orders.")
chargebacks = LlmAgent(name="chargebacks", model=M, tools=T, description="Chargebacks.")
refunds = LlmAgent(name="refunds", model=M, tools=T, sub_agents=[partner, chargebacks],
                   instruction="Partner-store orders: transfer to partner_refunds.")
app = App(name="refund_app", root_agent=refunds, plugins=[RefundGuardPlugin()])
ASKS = ["Refund $79 on A-100", "Partner-store order: refund $640 on A-200"]
async def main():
    async with InMemoryRunner(app=app) as runner:
        await runner.run_debug(ASKS, verbose=True)
asyncio.run(main())

Step 6 closes Lab 7: the $500 rule that the refund_agent instruction only asks for becomes code, moved from one agent into one plugin on the App. Step 1 shows the old way: a before_tool_callback on the refunds root agent guards that agent only, so partner_refunds and chargebacks can still refund 640 dollars on A-200. Steps 2 and 3 walk RefundGuardPlugin: its before_tool_callback gets keyword-only tool_args, and above 500 it returns a blocked dict, which skips the tool and becomes the tool result. Step 4 registers the plugin once with App(plugins=[...]) in guarded_app.py, where partner_refunds and chargebacks are sub-agents; plugins run before agent callbacks and the first non-None result wins, so a plugin block is final. With adk web you can load the same class with --extra_plugins and no code change. Compare the two: an agent callback covers one agent and gets args; a plugin hook covers every agent and tool in the App and gets tool_args. Think of a guard at the building door instead of a lock on one office. A likely question: "Do my agent callbacks still run?" Yes, after the plugins, unless a plugin already returned a result.
