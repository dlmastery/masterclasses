import asyncio
from typing import Literal
from google.adk import Agent, Context, Event, Workflow
from google.adk.workflow import node
from pydantic import BaseModel
class LineVerdict(BaseModel):
  line: str
  status: Literal["ok", "flag"]
  reason: str
auditor = Agent(name="auditor", model="gemini-3.5-flash",
    instruction=("You audit one expense line from the user message against policy: "
                 "meals <= $75/person, no alcohol, hotels <= $250/night. Return a verdict."),
    output_schema=LineVerdict)
@node(parameter_binding="node_input")
def policy_math(line: str, limit: float) -> bool:  # deterministic second opinion
  amount = float(line.rsplit("$", 1)[1])
  return amount <= limit
@node(rerun_on_resume=True)  # REQUIRED for any ctx.run_node caller
async def orchestrate(ctx: Context, node_input: str):
  lines = [l.strip() for l in node_input.split(";") if l.strip()]
  yield Event(message=f"Auditing {len(lines)} lines in parallel.")
  raw = await asyncio.gather(*[
      ctx.run_node(auditor, node_input=l, use_sub_branch=True) for l in lines])
  verdicts = [LineVerdict.model_validate(v) for v in raw]
  for v in verdicts:
    if v.status == "flag" and "hotel" in v.line.lower():
      ok = await ctx.run_node(policy_math, node_input={"line": v.line, "limit": 250.0})
      if ok:
        v.status, v.reason = "ok", "Within hotel cap per deterministic check."
  flagged = [v for v in verdicts if v.status == "flag"]
  yield Event(state={"flagged": [v.model_dump() for v in flagged]})
  yield Event(output=[v.model_dump() for v in verdicts],
    message="\n".join(f"{v.status.upper():4} {v.line} — {v.reason}" for v in verdicts))
root_agent = Workflow(name="expense_audit", edges=[("START", orchestrate)])

This is the orchestrator-workers pattern on an adjuster's travel expenses: the number of lines is known only at run time, so the graph cannot be static. The orchestrator splits the input, then starts one auditor run per line with ctx.run_node inside asyncio.gather, and use_sub_branch keeps the three runs from seeing each other's events. Think of a claims manager who hands each receipt to a different clerk and collects the slips. The hotel line shows a deterministic second opinion: policy_math checks 240 against the 250 cap in code, so an LLM mistake on a number cannot block a valid claim. A likely question is why rerun_on_resume is required: after an interruption the parent runs again from the top, and finished children are replayed from history, so put side effects in children and keep the parent thin. The codelab level L4b goes further with recursive spawning, bounded by a MAX_DEPTH check in code. Last build of the back office: subagents that research competitor premiums.
