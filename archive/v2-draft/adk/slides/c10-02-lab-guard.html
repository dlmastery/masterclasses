hr_agent/agent.py · part 1 of 2
import os
import re
from typing import Any, Optional
from google.adk import Agent
from google.adk.apps import App
from google.adk.integrations.model_armor import ModelArmorConfig, ModelArmorPlugin
from google.adk.models import LlmRequest, LlmResponse
from google.adk.plugins import BasePlugin
from google.adk.tools import BaseTool, ToolContext
from google.genai import types

SSN = re.compile(r"\b\d{3}-\d{2}-\d{4}\b")

class HrGuardPlugin(BasePlugin):
    def __init__(self) -> None:
        super().__init__(name="hr_guard")

    async def before_model_callback(self, *, callback_context,
                                    llm_request: LlmRequest) -> Optional[LlmResponse]:
        for content in llm_request.contents or []:
            for part in content.parts or []:
                if part.text and SSN.search(part.text):
                    # mutate, then continue
                    part.text = SSN.sub("[REDACTED-SSN]", part.text)
        return None

    async def before_tool_callback(self, *, tool: BaseTool, tool_args: dict[str, Any],
                                   tool_context: ToolContext) -> Optional[dict]:
        caller = tool_context.state.get("employee_id")
        if tool.name == "get_salary_band" and tool_args.get("employee_id") != caller:
            return {"status": "error",
                    "error_message": "You can only view your own salary band."}
        return None
hr_agent/agent.py · part 2 of 2
def get_salary_band(employee_id: str) -> dict:
    """Returns the salary band for an employee id such as E-1001."""
    return {"status": "success", "employee_id": employee_id,
            "band": "L5"}

root_agent = Agent(name="hr_agent", model="gemini-3.5-flash",
    instruction=("Answer Contoso HR policy questions. "
                 "Use get_salary_band only for the caller."),
    tools=[get_salary_band])

template = os.environ["MA_TEMPLATE"]
app = App(name="hr_agent", root_agent=root_agent, plugins=[
    HrGuardPlugin(),
    ModelArmorPlugin(config=ModelArmorConfig(
        prompt_template_name=template,
        response_template_name=template,
        tool_output_template_name=template,
        block_on_screening_failure=True)),
])
Rewrite each SSN to [REDACTED-SSN] in place. None means continue.
A returned dict replaces the tool result. Other employees' bands stay hidden.
tool_output_template_name (new) screens tool results. block_on_screening_failure=True is the default: fail closed.
The loader takes app before root_agent. Plugins run in list order. A reused name raises already registered.
'hr-guard' alone fails: ValueError: Template names must be full resource names. Keep all templates in one location.

This is step one of the Contoso HR lab: we put two independent guard layers in front of the model before the agent goes anywhere near production. Our own HrGuardPlugin is deterministic code: it rewrites any SSN in the request to REDACTED-SSN, and it refuses a salary lookup for any employee id that is not the caller's id in session state. The managed ModelArmorPlugin then screens the prompt, the reply and, new in this release, every tool output, which is the path that indirect injection uses; it fails closed by default when the Model Armor API errors. Think of it as an office building: the badge reader at the door is your code and knows exactly who may enter which room, and the security scanner inside is Model Armor, which looks for dangerous content but does not know your org chart. A likely question: "Why not let Model Armor do the salary check too?" Because authorization is a fixed rule that must hold every time, so it belongs in a tool guard, and a content classifier is the second layer, not the first.
