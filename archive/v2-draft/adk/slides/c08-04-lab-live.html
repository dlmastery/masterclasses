# imports: asyncio, wave, Agent, Runner, RunConfig,
#   LiveRequestQueue, InMemorySessionService, types
def get_balance(account_id: str) -> dict:
    """Returns balance and due date for an account."""
    return {"status": "success", "balance_usd": 84.20,
            "due": "2026-11-05"}
agent = Agent(name="voice_billing", tools=[get_balance],
    model="gemini-live-2.5-flash-native-audio",
    instruction="Be brief. Never quote new-plan prices.")
def save_wav(pcm: bytes) -> None:   # 24 kHz, mono
    with wave.open("answer.wav", "wb") as w:
        w.setparams((1, 2, 24000, 0, "NONE", None))
        w.writeframes(pcm)
live_billing.py · part 2 · the live loop
async def main() -> None:
    runner = Runner(agent=agent, app_name="bluefin_voice",
                    session_service=InMemorySessionService(),
                    auto_create_session=True)
    cfg = RunConfig(response_modalities=["AUDIO"],
        output_audio_transcription=types.AudioTranscriptionConfig(),
        session_resumption=types.SessionResumptionConfig())
    q = LiveRequestQueue()                 # one queue per call
    q.send_content(types.Content(role="user", parts=[
        types.Part(text="What's my balance on BF-3310?")]))
    pcm = bytearray()
    try:
        async for ev in runner.run_live(
                user_id="cust-42", session_id="call-1",
                live_request_queue=q, run_config=cfg):
            t = ev.output_transcription
            if t and t.text: print("agent:", t.text)
            for p in (ev.content.parts if ev.content else []):
                if p.inline_data: pcm += p.inline_data.data
            if ev.turn_complete: break
    finally: q.close()
    save_wav(bytes(pcm))
asyncio.run(main())

This is the last step of the Bluefin lab: the same support team now answers billing calls by voice through the Live API on Agent Platform, so the .env sets GOOGLE_GENAI_USE_ENTERPRISE=TRUE and a regional location; with an AI Studio key instead, set it to FALSE, add GOOGLE_API_KEY, and use the preview model gemini-3.1-flash-live-preview. Walk the code in four moves: a plain get_balance tool and an agent on the native-audio Live model; a Runner with auto_create_session, because run_live needs a session; a RunConfig that asks for audio out, an output transcript and session resumption; and a loop that reads events until turn_complete, closes the queue and saves the 24 kHz PCM as answer.wav so you can hear it. The script sends text as a stand-in for the microphone; a real call streams 16 kHz PCM chunks with send_realtime and adds input_audio_transcription. The LiveRequestQueue is like a phone line you hang up with close(), so you dial a new one for every call. Two traps: StreamingMode.BIDI in RunConfig does not start live mode, only runner.run_live does, and the location must be regional, not global. A likely question is "can I guard what the caller says?": model callbacks run in live mode in 2.13, but a before_model_callback sees only the transcript, after the model already heard the audio, so re-confirm account numbers in the tool, which is where the ASR paper says voice agents fail. A second paper, Heartbeat Thinking (arXiv 2610.26233), cut final-response latency by 36.4% by running read-only tool calls while the caller still speaks.
