from typing import Optional
from google.adk import Agent, Context, Event, Workflow
from google.adk.events import RequestInput
from google.adk.workflow import node
from pydantic import BaseModel
AUTO_APPROVE_LIMIT = 500.0
class ManagerDecision(BaseModel):
  approved: bool
  amount: Optional[float] = None          # partial refund
  note: str = ""
def intake(node_input: str):               # "A-2040 820"
  order_id, amount = node_input.split()[:2]
  yield Event(state={"order_id": order_id, "amount": float(amount), "note": ""})
draft_reply = Agent(name="draft_reply", output_key="draft",   # -> state["draft"]
    instruction="Write a refund email for ${amount} on {order_id}. Note: {note?}")
@node(rerun_on_resume=True)
def manager_gate(ctx: Context, amount: float, draft: str):    # args = state keys
  if amount <= AUTO_APPROVE_LIMIT:
    yield Event(route="approved", state={"final_amount": amount}); return
  iid = f"refund_approval_{ctx.invocation_id}_{ctx.run_id}"
  answer = ctx.resume_inputs.get(iid)
  if answer is None:
    yield RequestInput(interrupt_id=iid, response_schema=ManagerDecision,
                       payload={"amount": amount, "draft": draft},
                       message=f"Refund ${amount:.2f} needs approval.\n---\n{draft}")
    return
  d = ManagerDecision.model_validate(answer)
  if not d.approved: yield Event(route="rejected")
  elif d.note: yield Event(state={"note": d.note}, route="revise")
  else: yield Event(state={"final_amount": d.amount or amount}, route="approved")

This is step 1 of the Acme refunds lab: refunds up to $500 go straight through, and larger ones stop at manager_gate. Walk the four focus steps. First the data: intake splits "A-2040 820" and writes amount to state, draft_reply is an LLM Agent whose output_key writes draft, and manager_gate receives both as arguments because the parameter names match the state keys. Second, the limit: a $320 refund would route straight to issue_refund, which the dashed token shows. Third, the re-entry pattern: with rerun_on_resume=True the gate asks the question, the run pauses, and after the reply the whole function runs again from the top; this time resume_inputs.get(iid) finds the answer, because the id is built from invocation_id and run_id and both stay the same on a resume. With the default rerun_on_resume=False, the hand-off pattern, the reply itself becomes the node output and a next node must read it. Fourth, the answer routes to approved, rejected or revise; a revise pass runs the node again inside the same invocation, so run_id goes from 1 to 2 and the gate asks with a new id. Think of the gate as a sticky note on a form: the note keeps the same number, so when the form comes back the clerk finds the answer under that number. The .env line uses GOOGLE_GENAI_USE_ENTERPRISE, the current name; the older GOOGLE_GENAI_USE_VERTEXAI still works but raises a DeprecationWarning, and the key comes from AI Studio.
